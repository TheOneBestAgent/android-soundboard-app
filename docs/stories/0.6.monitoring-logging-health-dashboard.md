# Story 0.6: Monitoring, Logging & Health Dashboard - Brownfield Addition

## User Story

As a **developer and system administrator**,
I want **comprehensive monitoring, structured logging, and a health dashboard for the desktop server**,
So that **I can monitor system performance, debug issues quickly, and ensure the application is running optimally**.

## Story Context

**Existing System Integration:**

- Integrates with: All previous infrastructure stories (0.1-0.5)
- Technology: Winston for logging, Prometheus metrics, Express health endpoints
- Follows pattern: Standard Node.js monitoring with structured logging and metrics collection
- Touch points: All server components, database operations, external API calls, system resources

## Acceptance Criteria

**Functional Requirements:**

1. Structured logging with multiple levels and output formats
2. Application performance metrics collection and exposure
3. Health check endpoints for system status monitoring
4. Real-time system resource monitoring (CPU, memory, disk)
5. Database connection and query performance monitoring
6. External API call monitoring and error tracking

**Dashboard Requirements:**

7. Web-based health dashboard accessible via browser
8. Real-time metrics visualization with charts and graphs
9. System alerts and notifications for critical issues
10. Log viewing and filtering interface
11. Performance trends and historical data analysis

**Integration Requirements:**

12. Logging integrates with configuration management (Story 0.5)
13. Metrics collection works with all server endpoints (Story 0.1)
14. Database monitoring tracks Prisma operations (Story 0.2)
15. Electron app status is included in monitoring (Story 0.3)
16. Build pipeline includes monitoring setup (Story 0.4)

## Technical Notes

- **Integration Approach:** Middleware-based monitoring with centralized logging and metrics collection
- **Existing Pattern Reference:** Standard observability stack with custom dashboard
- **Key Constraints:** Minimal performance impact, configurable verbosity, secure access

**Minimal Risk Assessment:**

- **Primary Risk:** Performance overhead from extensive monitoring
- **Mitigation:** Configurable monitoring levels, efficient data collection
- **Rollback:** Disable monitoring features while maintaining basic logging

**Compatibility Verification:**

- [ ] Monitoring works across all target platforms
- [ ] No conflicts with existing application performance
- [ ] Dashboard is accessible from development and production environments
- [ ] Metrics collection is efficient and non-blocking

## Winston Logger Configuration

```typescript
// src/monitoring/logger.ts
import winston from 'winston';
import DailyRotateFile from 'winston-daily-rotate-file';
import { appConfig } from '../config';

const logLevel = appConfig.get('logging.level');
const logDir = appConfig.get('logging.filePath').replace('/app.log', '');

// Custom log format
const logFormat = winston.format.combine(
  winston.format.timestamp({
    format: 'YYYY-MM-DD HH:mm:ss'
  }),
  winston.format.errors({ stack: true }),
  winston.format.json(),
  winston.format.printf(({ timestamp, level, message, stack, ...meta }) => {
    let log = `${timestamp} [${level.toUpperCase()}]: ${message}`;
    
    if (Object.keys(meta).length > 0) {
      log += ` ${JSON.stringify(meta)}`;
    }
    
    if (stack) {
      log += `\n${stack}`;
    }
    
    return log;
  })
);

// Sensitive data masking
const maskSensitiveData = winston.format((info) => {
  const sensitiveFields = ['password', 'token', 'secret', 'key', 'authorization'];
  
  function maskObject(obj: any): any {
    if (typeof obj !== 'object' || obj === null) return obj;
    
    const masked = { ...obj };
    for (const [key, value] of Object.entries(masked)) {
      if (sensitiveFields.some(field => key.toLowerCase().includes(field))) {
        masked[key] = '***MASKED***';
      } else if (typeof value === 'object') {
        masked[key] = maskObject(value);
      }
    }
    return masked;
  }
  
  return maskObject(info);
});

// Create logger instance
export const logger = winston.createLogger({
  level: logLevel,
  format: winston.format.combine(
    maskSensitiveData(),
    logFormat
  ),
  transports: [
    // Console transport for development
    new winston.transports.Console({
      format: winston.format.combine(
        winston.format.colorize(),
        winston.format.simple()
      ),
      silent: process.env.NODE_ENV === 'production'
    }),
    
    // Application logs
    new DailyRotateFile({
      filename: `${logDir}/app-%DATE%.log`,
      datePattern: 'YYYY-MM-DD',
      maxSize: appConfig.get('logging.maxSize'),
      maxFiles: appConfig.get('logging.maxFiles'),
      level: 'info'
    }),
    
    // Error logs
    new DailyRotateFile({
      filename: `${logDir}/error-%DATE%.log`,
      datePattern: 'YYYY-MM-DD',
      maxSize: appConfig.get('logging.maxSize'),
      maxFiles: appConfig.get('logging.maxFiles'),
      level: 'error'
    }),
    
    // Performance logs
    new DailyRotateFile({
      filename: `${logDir}/performance-%DATE%.log`,
      datePattern: 'YYYY-MM-DD',
      maxSize: appConfig.get('logging.maxSize'),
      maxFiles: appConfig.get('logging.maxFiles'),
      level: 'debug',
      format: winston.format.combine(
        winston.format.label({ label: 'PERFORMANCE' }),
        logFormat
      )
    })
  ]
});

// Performance logging helper
export function logPerformance(operation: string, duration: number, metadata?: any) {
  logger.debug('Performance metric', {
    operation,
    duration,
    timestamp: Date.now(),
    ...metadata
  });
}

// Request logging middleware
export function requestLogger() {
  return (req: any, res: any, next: any) => {
    const start = Date.now();
    
    res.on('finish', () => {
      const duration = Date.now() - start;
      logger.info('HTTP Request', {
        method: req.method,
        url: req.url,
        statusCode: res.statusCode,
        duration,
        userAgent: req.get('User-Agent'),
        ip: req.ip
      });
    });
    
    next();
  };
}
```

```typescript
  name: 'discord_api_calls_total',
  help: 'Total number of Discord API calls',
  labelNames: ['endpoint', 'status']
});

export const discordApiDuration = new Histogram({
  name: 'discord_api_duration_seconds',
  help: 'Duration of Discord API calls',
  labelNames: ['endpoint'],
  buckets: [0.1, 0.5, 1, 2, 5]
});

// Error tracking
export const applicationErrors = new Counter({
  name: 'application_errors_total',
  help: 'Total number of application errors',
  labelNames: ['type', 'severity']
});

// Middleware for HTTP metrics
export function metricsMiddleware() {
  return (req: any, res: any, next: any) => {
    const start = Date.now();
    
    res.on('finish', () => {
      const duration = (Date.now() - start) / 1000;
      const route = req.route?.path || req.path;
      
      httpRequestDuration
        .labels(req.method, route, res.statusCode.toString())
        .observe(duration);
      
      httpRequestTotal
        .labels(req.method, route, res.statusCode.toString())
        .inc();
    });
    
    next();
  };
}

// System metrics collection
export async function collectSystemMetrics() {
  try {
    const si = await import('systeminformation');
    
    // Memory metrics
    const memory = await si.mem();
    systemMemoryUsage.labels('total').set(memory.total);
    systemMemoryUsage.labels('used').set(memory.used);
    systemMemoryUsage.labels('free').set(memory.free);
    
    // CPU metrics
    const cpu = await si.currentLoad();
    systemCpuUsage.set(cpu.currentLoad);
    
  } catch (error) {
    logger.error('Failed to collect system metrics', { error });
  }
}

// Start system metrics collection
setInterval(collectSystemMetrics, 30000); // Every 30 seconds

export { register };
```

## Health Check Implementation

```typescript
// src/monitoring/health.ts
import { Request, Response } from 'express';
import { PrismaClient } from '@prisma/client';
import { logger } from './logger';
import { appConfig } from '../config';

interface HealthCheck {
  name: string;
  status: 'healthy' | 'unhealthy' | 'degraded';
  message?: string;
  responseTime?: number;
  details?: any;
}

interface HealthStatus {
  status: 'healthy' | 'unhealthy' | 'degraded';
  timestamp: string;
  uptime: number;
  version: string;
  checks: HealthCheck[];
}

const prisma = new PrismaClient();

// Database health check
async function checkDatabase(): Promise<HealthCheck> {
  const start = Date.now();
  
  try {
    await prisma.$queryRaw`SELECT 1`;
    const responseTime = Date.now() - start;
    
    return {
      name: 'database',
      status: responseTime < 1000 ? 'healthy' : 'degraded',
      responseTime,
      message: responseTime < 1000 ? 'Database is responsive' : 'Database is slow'
    };
  } catch (error) {
    return {
      name: 'database',
      status: 'unhealthy',
      responseTime: Date.now() - start,
      message: 'Database connection failed',
      details: error instanceof Error ? error.message : 'Unknown error'
    };
  }
}

// Memory health check
function checkMemory(): HealthCheck {
  const memUsage = process.memoryUsage();
  const totalMem = memUsage.heapTotal;
  const usedMem = memUsage.heapUsed;
  const memPercent = (usedMem / totalMem) * 100;
  
  let status: 'healthy' | 'unhealthy' | 'degraded' = 'healthy';
  let message = 'Memory usage is normal';
  
  if (memPercent > 90) {
    status = 'unhealthy';
    message = 'Memory usage is critically high';
  } else if (memPercent > 75) {
    status = 'degraded';
    message = 'Memory usage is elevated';
  }
  
  return {
    name: 'memory',
    status,
    message,
    details: {
      heapUsed: Math.round(usedMem / 1024 / 1024),
      heapTotal: Math.round(totalMem / 1024 / 1024),
      percentage: Math.round(memPercent)
    }
  };
}

// Disk space health check
async function checkDiskSpace(): Promise<HealthCheck> {
  try {
    const si = await import('systeminformation');
    const disks = await si.fsSize();
    
    const mainDisk = disks[0];
    const usedPercent = (mainDisk.used / mainDisk.size) * 100;
    
    let status: 'healthy' | 'unhealthy' | 'degraded' = 'healthy';
    let message = 'Disk space is sufficient';
    
    if (usedPercent > 95) {
      status = 'unhealthy';
      message = 'Disk space is critically low';
    } else if (usedPercent > 85) {
      status = 'degraded';
      message = 'Disk space is running low';
    }
    
    return {
      name: 'disk',
      status,
      message,
      details: {
        used: Math.round(mainDisk.used / 1024 / 1024 / 1024),
        total: Math.round(mainDisk.size / 1024 / 1024 / 1024),
        percentage: Math.round(usedPercent)
      }
    };
  } catch (error) {
    return {
      name: 'disk',
      status: 'unhealthy',
      message: 'Failed to check disk space',
      details: error instanceof Error ? error.message : 'Unknown error'
    };
  }
}

// External services health check
async function checkExternalServices(): Promise<HealthCheck> {
  // This would check Discord API, etc.
  // For now, return a placeholder
  return {
    name: 'external_services',
    status: 'healthy',
    message: 'External services are accessible'
  };
}

// Main health check function
export async function performHealthCheck(): Promise<HealthStatus> {
  const startTime = Date.now();
  
  try {
    const checks = await Promise.all([
      checkDatabase(),
      checkMemory(),
      checkDiskSpace(),
      checkExternalServices()
    ]);
    
    // Determine overall status
    const hasUnhealthy = checks.some(check => check.status === 'unhealthy');
    const hasDegraded = checks.some(check => check.status === 'degraded');
    
    let overallStatus: 'healthy' | 'unhealthy' | 'degraded' = 'healthy';
    if (hasUnhealthy) {
      overallStatus = 'unhealthy';
    } else if (hasDegraded) {
      overallStatus = 'degraded';
    }
    
    const healthStatus: HealthStatus = {
      status: overallStatus,
      timestamp: new Date().toISOString(),
      uptime: process.uptime(),
      version: process.env.npm_package_version || '1.0.0',
      checks
    };
    
    logger.info('Health check completed', {
      status: overallStatus,
      duration: Date.now() - startTime,
      checksCount: checks.length
    });
    
    return healthStatus;
    
  } catch (error) {
    logger.error('Health check failed', { error });
    
    return {
      status: 'unhealthy',
      timestamp: new Date().toISOString(),
      uptime: process.uptime(),
      version: process.env.npm_package_version || '1.0.0',
      checks: [{
        name: 'health_check',
        status: 'unhealthy',
        message: 'Health check system failed',
        details: error instanceof Error ? error.message : 'Unknown error'
      }]
    };
  }
}

// Health check endpoint handlers
export async function healthCheckHandler(req: Request, res: Response) {
  const health = await performHealthCheck();
  
  const statusCode = health.status === 'healthy' ? 200 : 
                    health.status === 'degraded' ? 200 : 503;
  
  res.status(statusCode).json(health);
}

export function livenessHandler(req: Request, res: Response) {
  res.status(200).json({
    status: 'alive',
    timestamp: new Date().toISOString(),
    uptime: process.uptime()
  });
}

export function readinessHandler(req: Request, res: Response) {
  // Simple readiness check - server is ready if it can respond
  res.status(200).json({
    status: 'ready',
    timestamp: new Date().toISOString()
  });
}
```
