## Story

**I want** a comprehensive health dashboard that displays real-time system metrics and performance data,
**so that** I can quickly identify and resolve audio latency, connectivity, or resource usage problems during live streaming.

5. Audio device health and configuration status
6. Historical performance data with 24-hour trend charts
7. Quick diagnostic tools for common issues
8. Export functionality for performance reports
9. Alert system for critical performance thresholds
10. Mobile-optimized dashboard layout with key metrics
11. Accessibility support with screen reader compatibility
12. Integration with system notifications for critical alerts

### Metrics Collection System
```typescript
// Real-time metrics collection and aggregation
class HealthMonitoringService {
  private metricsCollectors: Map<string, MetricsCollector> = new Map();
  private alertEngine: AlertEngine;
  private metricsStore: MetricsStore;
  private isMonitoring = false;
  private collectionInterval = 1000; // 1 second
  
  constructor() {
    this.alertEngine = new AlertEngine();
    this.metricsStore = new MetricsStore();
    this.initializeCollectors();
  }
  
  private initializeCollectors() {
    // System metrics collector
    this.metricsCollectors.set('system', new SystemMetricsCollector());
    
    // Audio metrics collector
    this.metricsCollectors.set('audio', new AudioMetricsCollector());
    
    // Network metrics collector
    this.metricsCollectors.set('network', new NetworkMetricsCollector());
    
    // Application metrics collector
    this.metricsCollectors.set('application', new ApplicationMetricsCollector());
  }
  
  async startMonitoring(): Promise<void> {
    if (this.isMonitoring) return;
    
    this.isMonitoring = true;
    
    // Start collection loop
    this.collectionLoop();
    
    // Initialize baseline calculations
    await this.calculateBaselines();
    
    console.log('Health monitoring started');
  }
  
  async stopMonitoring(): Promise<void> {
    this.isMonitoring = false;
    console.log('Health monitoring stopped');
  }
  
  private async collectionLoop(): Promise<void> {
    while (this.isMonitoring) {
      try {
        const startTime = Date.now();
        
        // Collect metrics from all collectors
        const metricsPromises = Array.from(this.metricsCollectors.entries()).map(
          async ([category, collector]) => {
            try {
              const metrics = await collector.collect();
              return { category, metrics };
            } catch (error) {
              console.error(`Failed to collect ${category} metrics:`, error);
              return null;
            }
          }
        );
        
        const results = await Promise.all(metricsPromises);
        
        // Aggregate metrics
        const aggregatedMetrics: SystemMetrics = {
          id: generateId(),
          timestamp: new Date(),
          deviceId: await this.getDeviceId(),
          performance: results.find(r => r?.category === 'system')?.metrics || {},
          audio: results.find(r => r?.category === 'audio')?.metrics || {},
          network: results.find(r => r?.category === 'network')?.metrics || {},
          application: results.find(r => r?.category === 'application')?.metrics || {},
        };
        
        // Store metrics
        await this.metricsStore.store(aggregatedMetrics);
        
        // Check for alerts
        await this.alertEngine.checkThresholds(aggregatedMetrics);
        
        // Notify subscribers
        this.notifyMetricsUpdate(aggregatedMetrics);
        
        // Calculate next collection time
        const collectionTime = Date.now() - startTime;
        const nextInterval = Math.max(0, this.collectionInterval - collectionTime);
        
        await new Promise(resolve => setTimeout(resolve, nextInterval));
      } catch (error) {
        console.error('Metrics collection error:', error);
        await new Promise(resolve => setTimeout(resolve, this.collectionInterval));
      }
    }
  }
  
  async getMetricsHistory(
    category: string,
    timeRange: { start: Date; end: Date }
  ): Promise<SystemMetrics[]> {
    return this.metricsStore.getHistory(category, timeRange);
  }
  
  async generatePerformanceReport(
    timeRange: { start: Date; end: Date }
  ): Promise<PerformanceReport> {
    const metrics = await this.metricsStore.getHistory('all', timeRange);
    
    return {
      id: generateId(),
      timeRange,
      summary: {
        totalSamples: metrics.length,
        averageLatency: this.calculateAverage(metrics, 'audio.playbackLatency'),
        peakCpuUsage: this.calculateMax(metrics, 'performance.cpu.usage'),
        peakMemoryUsage: this.calculateMax(metrics, 'performance.memory.usage'),
        networkUptime: this.calculateUptime(metrics, 'network.connectionType'),
        alertCount: await this.alertEngine.getAlertCount(timeRange),
      },
      trends: {
        latencyTrend: this.calculateTrend(metrics, 'audio.playbackLatency'),
        cpuTrend: this.calculateTrend(metrics, 'performance.cpu.usage'),
        memoryTrend: this.calculateTrend(metrics, 'performance.memory.usage'),
      },
      recommendations: await this.generateRecommendations(metrics),
      generatedAt: new Date(),
    };
  }
  
  private async calculateBaselines(): Promise<void> {
    // Calculate performance baselines for the device
    const historicalData = await this.metricsStore.getHistory('all', {
      start: new Date(Date.now() - 7 * 24 * 60 * 60 * 1000), // 7 days
      end: new Date(),
    });
    
    if (historicalData.length < 100) {
      console.log('Insufficient data for baseline calculation');
      return;
    }
    
    const baselines = {
      audio: this.calculateMetricBaseline(historicalData, 'audio'),
      system: this.calculateMetricBaseline(historicalData, 'performance'),
      network: this.calculateMetricBaseline(historicalData, 'network'),
    };
    
    await this.metricsStore.storeBaselines(baselines);
  }
}

class SystemMetricsCollector implements MetricsCollector {
  async collect(): Promise<any> {
    try {
      const systemMetrics = await SystemMetricsModule.getCurrentMetrics();
      
      return {
        cpu: {
          usage: systemMetrics.cpu.usage,
          cores: systemMetrics.cpu.cores,
          frequency: systemMetrics.cpu.frequency,
          temperature: systemMetrics.cpu.temperature,
        },
        memory: {
          used: systemMetrics.memory.used,
          total: systemMetrics.memory.total,
          available: systemMetrics.memory.available,

        },
        disk: {
          used: systemMetrics.disk.used,
          total: systemMetrics.disk.total,
          available: systemMetrics.disk.available,

          writeSpeed: systemMetrics.disk.writeSpeed,
        },
      };
    } catch (error) {
      console.error('System metrics collection failed:', error);

    }
  }
```

### Alert Engine
```typescript
// Intelligent alerting system with threshold management
class AlertEngine {
  private alertRules: Map<string, AlertRule> = new Map();
  private activeAlerts: Map<string, HealthAlert> = new Map();
  private notificationService: NotificationService;
  
  constructor() {
    this.notificationService = new NotificationService();
    this.initializeDefaultRules();
  }
  
  private initializeDefaultRules() {
    // Audio latency alert
    this.alertRules.set('audio_latency_high', {
      id: 'audio_latency_high',
      name: 'High Audio Latency',
      category: 'audio',
      metric: 'audio.playbackLatency',
      operator: '>',
      threshold: 10, // milliseconds
      severity: 4,
      enabled: true,
      cooldown: 30000, // 30 seconds
    });
    
    // CPU usage alert
    this.alertRules.set('cpu_usage_high', {
      id: 'cpu_usage_high',
      name: 'High CPU Usage',
      category: 'performance',
      metric: 'performance.cpu.usage',
      operator: '>',
      threshold: 80, // percent
      severity: 3,
      enabled: true,
      cooldown: 60000, // 1 minute
    });
    
    // Memory usage alert
    this.alertRules.set('memory_usage_high', {
      id: 'memory_usage_high',
      name: 'High Memory Usage',
      category: 'performance',
      metric: 'performance.memory.usage',
      operator: '>',
      threshold: 85, // percent
      severity: 3,
      enabled: true,
      cooldown: 60000,
    });
    
    // Network connectivity alert
    this.alertRules.set('network_disconnected', {
      id: 'network_disconnected',
      name: 'Network Disconnected',
      category: 'network',
      metric: 'network.connectionType',
      operator: '==',
      threshold: 'offline',
      severity: 5,
      enabled: true,
      cooldown: 10000, // 10 seconds
    });
    
    // Audio device disconnected
    this.alertRules.set('audio_device_disconnected', {
      id: 'audio_device_disconnected',
      name: 'Audio Device Disconnected',
      category: 'audio',
      metric: 'audio.deviceStatus',
      operator: '==',
      threshold: 'disconnected',
      severity: 4,
      enabled: true,
      cooldown: 5000, // 5 seconds
    });
  }
  
  async checkThresholds(metrics: SystemMetrics): Promise<void> {
    for (const [ruleId, rule] of this.alertRules) {
      if (!rule.enabled) continue;
      
      try {
        const currentValue = this.getMetricValue(metrics, rule.metric);
        const shouldAlert = this.evaluateThreshold(currentValue, rule);
        
        if (shouldAlert) {
          await this.triggerAlert(rule, currentValue, metrics);
        } else {
          await this.resolveAlert(ruleId);
        }
      } catch (error) {
        console.error(`Alert rule evaluation failed for ${ruleId}:`, error);
      }
    }
  }
  
  private async triggerAlert(
    rule: AlertRule,
    currentValue: any,
    metrics: SystemMetrics
  ): Promise<void> {
    const existingAlert = this.activeAlerts.get(rule.id);
    
    // Check cooldown period
    if (existingAlert && 
        Date.now() - existingAlert.triggeredAt.getTime() < rule.cooldown) {
      return;
    }
    
    const alert: HealthAlert = {
      id: generateId(),
      type: this.getSeverityType(rule.severity),
      category: rule.category,
      title: rule.name,
      description: this.generateAlertDescription(rule, currentValue),
      threshold: {
        metric: rule.metric,
        operator: rule.operator,
        value: rule.threshold,
        unit: this.getMetricUnit(rule.metric),
      },
      currentValue,
      severity: rule.severity,
      isActive: true,
      triggeredAt: new Date(),
      actions: this.generateAlertActions(rule, metrics),
    };
    
    this.activeAlerts.set(rule.id, alert);
    
    // Send notification
    await this.notificationService.sendAlert(alert);
    
    // Store alert in database
    await this.storeAlert(alert);
    
    console.log(`Alert triggered: ${alert.title}`);
  }
  
  private async resolveAlert(ruleId: string): Promise<void> {
    const alert = this.activeAlerts.get(ruleId);
    if (!alert || !alert.isActive) return;
    
    alert.isActive = false;
    alert.resolvedAt = new Date();
    
    this.activeAlerts.delete(ruleId);
    
    // Send resolution notification
    await this.notificationService.sendAlertResolution(alert);
    
    // Update alert in database
    await this.updateAlert(alert);
    
    console.log(`Alert resolved: ${alert.title}`);
  }
  
  private generateAlertActions(rule: AlertRule, metrics: SystemMetrics): any[] {
    const actions = [];
    
    switch (rule.id) {
      case 'audio_latency_high':
        actions.push({
          id: 'restart_audio_engine',
          title: 'Restart Audio Engine',
          description: 'Restart the audio engine to reduce latency',
          type: 'manual',
        });
        actions.push({
          id: 'adjust_buffer_size',
          title: 'Adjust Buffer Size',
          description: 'Increase audio buffer size for stability',
          type: 'automatic',
          command: 'audio.adjustBufferSize(512)',
        });
        break;
        
      case 'cpu_usage_high':
        actions.push({
          id: 'close_background_apps',
          title: 'Close Background Apps',
          description: 'Close unnecessary background applications',
          type: 'manual',
        });
        actions.push({
          id: 'reduce_audio_quality',
          title: 'Reduce Audio Quality',
          description: 'Temporarily reduce audio processing quality',
          type: 'automatic',
          command: 'audio.setQuality("medium")',
        });
        break;
        
      case 'memory_usage_high':
        actions.push({
          id: 'clear_cache',
          title: 'Clear Cache',
          description: 'Clear application cache to free memory',
          type: 'automatic',
          command: 'cache.clear()',
        });
        break;
        
      case 'network_disconnected':
        actions.push({
          id: 'check_network_settings',
          title: 'Check Network Settings',
          description: 'Verify network configuration and connectivity',
          type: 'manual',
        });
        actions.push({
          id: 'enable_offline_mode',
          title: 'Enable Offline Mode',
          description: 'Switch to offline mode for local functionality',
          type: 'automatic',
          command: 'app.setOfflineMode(true)',
        });
        break;
    }
    
    return actions;
  }
}
```

```typescript
          datasets: [{
            data: chartData.map(d => d.y),
            color: () => color,
            strokeWidth: 2,
          }],
        }}
        width={Dimensions.get('window').width - 40}
        height={200}
        chartConfig={{
          backgroundColor: 'transparent',
          backgroundGradientFrom: '#ffffff',
          backgroundGradientTo: '#ffffff',
          decimalPlaces: 1,
          color: (opacity = 1) => `rgba(0, 0, 0, ${opacity})`,
          style: {
            borderRadius: 16,
          },
        }}
        bezier
        style={styles.chart}
      />
      
      {/* Threshold line */}
      {threshold && (
        <View
          style={[
            styles.thresholdLine,
            {
              bottom: (threshold / Math.max(...chartData.map(d => d.y))) * 200,
            },
          ]}
        />
      )}
    </View>
  );
};
```
