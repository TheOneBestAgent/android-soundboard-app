### Story
As a content creator and streamer using the soundboard app, I need seamless integration with OBS Studio via WebSocket, so that I can control scenes, sources, and streaming functions directly from my soundboard, enhancing my live streaming workflow and audience engagement.

### Acceptance Criteria
- **OBS WebSocket Connection**: Establish and maintain WebSocket connection to OBS Studio (default ws://localhost:4455)
- **Scene Management**: Get scene list, switch between scenes, and monitor current scene status
- **Source Control**: Control audio sources, show/hide sources, and adjust source properties
- **Stream Control**: Start/stop streaming, get stream status, and monitor stream health
- **Recording Control**: Start/stop recording, get recording status, and manage recording settings
- **Audio Integration**: Route soundboard audio to OBS as audio source with real-time mixing
- **Trigger Automation**: Automatically trigger OBS actions based on soundboard button presses
- **Connection Management**: Auto-reconnect on connection loss, connection status monitoring
- **Error Handling**: Graceful handling of OBS disconnection, invalid commands, and API errors
- **Performance**: OBS commands execute within 200ms, scene changes within 100ms
- **Security**: Secure WebSocket authentication with password protection
- **Configuration**: User-friendly OBS connection setup and settings management

#### OBS WebSocket Client Implementation
```typescript
// Comprehensive OBS WebSocket client for React Native
class OBSWebSocketClient {
  private ws: WebSocket | null = null;
  private connection: OBSConnection;
  private eventHandlers: Map<string, Function[]> = new Map();
  private reconnectTimer: NodeJS.Timeout | null = null;
  private heartbeatTimer: NodeJS.Timeout | null = null;
  private requestId = 0;
  private pendingRequests: Map<string, { resolve: Function; reject: Function; timeout: NodeJS.Timeout }> = new Map();
  
  constructor(connection: OBSConnection) {
    this.connection = connection;
  }
  
  async connect(): Promise<void> {
    try {
      const wsUrl = `ws://${this.connection.host}:${this.connection.port}`;
      console.log(`Connecting to OBS at ${wsUrl}`);
      
      this.ws = new WebSocket(wsUrl, 'obs-websocket');
      
      this.ws.onopen = () => {
        console.log('OBS WebSocket connected');
        this.connection.connectionStatus = 'connected';
        this.connection.isConnected = true;
        this.connection.lastConnected = new Date();
        
        // Start authentication if password is provided
        if (this.connection.password) {
          this.authenticate();
        }
        
        // Start heartbeat
        this.startHeartbeat();
        
        this.emit('connected', this.connection);
      };
      
      this.ws.onmessage = (event) => {
        this.handleMessage(JSON.parse(event.data));
      };
      
      this.ws.onclose = (event) => {
        console.log('OBS WebSocket disconnected:', event.code, event.reason);
        this.connection.connectionStatus = 'disconnected';
        this.connection.isConnected = false;
        
        this.stopHeartbeat();
        this.emit('disconnected', { code: event.code, reason: event.reason });
        
        // Auto-reconnect if not intentionally closed
        if (event.code !== 1000) {
          this.scheduleReconnect();
        }
      };
      
      this.ws.onerror = (error) => {
        console.error('OBS WebSocket error:', error);
        this.connection.connectionStatus = 'error';
        this.emit('error', error);
      };
      
    } catch (error) {
      console.error('Failed to connect to OBS:', error);
      this.connection.connectionStatus = 'error';
      throw error;
    }
  }
  
  async disconnect(): Promise<void> {
    if (this.reconnectTimer) {
      clearTimeout(this.reconnectTimer);
      this.reconnectTimer = null;
    }
    
    this.stopHeartbeat();
    
    if (this.ws && this.ws.readyState === WebSocket.OPEN) {
      this.ws.close(1000, 'Client disconnect');
    }
    
    this.connection.connectionStatus = 'disconnected';
    this.connection.isConnected = false;
  }
  
  private async authenticate(): Promise<void> {
    try {
      // OBS WebSocket v5 authentication flow
      const authResponse = await this.request('GetAuthRequired');
      
      if (authResponse.authRequired) {
        const { salt, challenge } = authResponse;
        
        // Generate authentication response
        const secret = await this.generateAuthSecret(this.connection.password!, salt);
        const authString = await this.generateAuthResponse(secret, challenge);
        
        await this.request('Authenticate', { auth: authString });
        console.log('OBS authentication successful');
      }
    } catch (error) {
      console.error('OBS authentication failed:', error);
      throw error;
    }
  }
  
  private async generateAuthSecret(password: string, salt: string): Promise<string> {
    const encoder = new TextEncoder();
    const passwordData = encoder.encode(password + salt);
    
    const hashBuffer = await crypto.subtle.digest('SHA-256', passwordData);
    const hashArray = Array.from(new Uint8Array(hashBuffer));
    
    return btoa(String.fromCharCode.apply(null, hashArray));
  }
  
  private async generateAuthResponse(secret: string, challenge: string): Promise<string> {
    const encoder = new TextEncoder();
    const responseData = encoder.encode(secret + challenge);
    
    const hashBuffer = await crypto.subtle.digest('SHA-256', responseData);
    const hashArray = Array.from(new Uint8Array(hashBuffer));
    
    return btoa(String.fromCharCode.apply(null, hashArray));
  }
  
  async request(requestType: string, requestData?: any): Promise<any> {
    return new Promise((resolve, reject) => {
      if (!this.ws || this.ws.readyState !== WebSocket.OPEN) {
        reject(new Error('OBS WebSocket not connected'));
        return;
      }
      
      const requestId = `req_${++this.requestId}`;
      
      const message = {
        op: 6, // Request
        d: {
          requestType,
          requestId,
          requestData: requestData || {},
        },
      };
      
      // Set timeout for request
      const timeout = setTimeout(() => {
        this.pendingRequests.delete(requestId);
        reject(new Error(`OBS request timeout: ${requestType}`));
      }, 5000);
      
      this.pendingRequests.set(requestId, { resolve, reject, timeout });
      
      this.ws.send(JSON.stringify(message));
    });
  }
  
  private handleMessage(message: any): void {
    switch (message.op) {
      case 7: // RequestResponse
        this.handleRequestResponse(message.d);
        break;
      case 5: // Event
        this.handleEvent(message.d);
        break;
      case 2: // Identify
        this.handleIdentify(message.d);
        break;
      default:
        console.log('Unknown OBS message:', message);
    }
  }
  
  private handleRequestResponse(data: any): void {
    const { requestId, requestStatus, responseData } = data;
    const pending = this.pendingRequests.get(requestId);
    
    if (pending) {
      clearTimeout(pending.timeout);
      this.pendingRequests.delete(requestId);
      
      if (requestStatus.result) {
        pending.resolve(responseData);
      } else {
        pending.reject(new Error(`OBS request failed: ${requestStatus.comment}`));
      }
    }
  }
  
  private handleEvent(eventData: any): void {
    const { eventType, eventData: data } = eventData;
    
    console.log(`OBS Event: ${eventType}`, data);
    
    // Emit event to listeners
    this.emit(eventType, data);
    
    // Handle specific events
    switch (eventType) {
      case 'CurrentProgramSceneChanged':
        this.emit('sceneChanged', data.sceneName);
        break;
      case 'StreamStateChanged':
        this.emit('streamStatusChanged', data);
        break;
      case 'RecordStateChanged':
        this.emit('recordingStatusChanged', data);
        break;
      case 'SourceVolumeChanged':
        this.emit('sourceVolumeChanged', data);
        break;
    }
  }
  
  private startHeartbeat(): void {
    this.heartbeatTimer = setInterval(() => {
      if (this.ws && this.ws.readyState === WebSocket.OPEN) {
        this.ws.send(JSON.stringify({ op: 3 })); // Heartbeat
      }
    }, 10000); // 10 seconds
  }
  
  private stopHeartbeat(): void {
    if (this.heartbeatTimer) {
      clearInterval(this.heartbeatTimer);
      this.heartbeatTimer = null;
    }
  }
  
  private scheduleReconnect(): void {
    if (this.reconnectTimer) return;
    
    console.log('Scheduling OBS reconnect in 5 seconds...');
    this.reconnectTimer = setTimeout(() => {
      this.reconnectTimer = null;
      this.connect().catch(error => {
        console.error('OBS reconnect failed:', error);
        this.scheduleReconnect(); // Try again
      });
    }, 5000);
  }
  
  on(event: string, handler: Function): void {
    if (!this.eventHandlers.has(event)) {
      this.eventHandlers.set(event, []);
    }
    this.eventHandlers.get(event)!.push(handler);
  }
  
  off(event: string, handler: Function): void {
    const handlers = this.eventHandlers.get(event);
    if (handlers) {
      const index = handlers.indexOf(handler);
      if (index > -1) {
        handlers.splice(index, 1);
      }
    }
  }
  
  private emit(event: string, data?: any): void {
    const handlers = this.eventHandlers.get(event);
    if (handlers) {
      handlers.forEach(handler => {
        try {
          handler(data);
        } catch (error) {
          console.error(`Error in OBS event handler for ${event}:`, error);
        }
      });
    }
  }
  
  // High-level OBS operations
  async getSceneList(): Promise<OBSScene[]> {
    const response = await this.request('GetSceneList');
    return response.scenes.map((scene: any) => ({
      id: scene.sceneUuid || scene.sceneName,
      name: scene.sceneName,
      isActive: scene.sceneName === response.currentProgramSceneName,
      sources: scene.sceneItems || [],
      transitions: [],
      previewMode: false,
      lastModified: new Date(),
    }));
  }
  
  async setCurrentScene(sceneName: string): Promise<void> {
    await this.request('SetCurrentProgramScene', { sceneName });
  }
  
  async getStreamStatus(): Promise<OBSStreamStatus> {
    const response = await this.request('GetStreamStatus');
    return {
      isStreaming: response.outputActive,
      isRecording: false, // Will be updated with recording status
      streamTime: response.outputDuration || 0,
      recordingTime: 0,
      bytesPerSecond: response.outputBytes || 0,
      kbitsPerSecond: (response.outputBytes || 0) * 8 / 1000,
      strain: 0,
      totalFrames: response.outputTotalFrames || 0,
      droppedFrames: response.outputSkippedFrames || 0,
      fps: 0,
    };
  }
  
  async startStream(): Promise<void> {
    await this.request('StartStream');
  }
  
  async stopStream(): Promise<void> {
    await this.request('StopStream');
  }
  
  async startRecording(): Promise<void> {
    await this.request('StartRecord');
  }
  
  async stopRecording(): Promise<void> {
    await this.request('StopRecord');
  }
  
  async setSourceVisibility(sourceName: string, visible: boolean): Promise<void> {
    await this.request('SetSceneItemEnabled', {
      sceneName: await this.getCurrentSceneName(),
      sceneItemId: await this.getSourceId(sourceName),
      sceneItemEnabled: visible,
    });
  }
  
  async setSourceVolume(sourceName: string, volume: number): Promise<void> {
    await this.request('SetInputVolume', {
      inputName: sourceName,
      inputVolumeMul: volume,
    });
  }
  
  private async getCurrentSceneName(): Promise<string> {
    const response = await this.request('GetCurrentProgramScene');
    return response.currentProgramSceneName;
  }
  
  private async getSourceId(sourceName: string): Promise<number> {
    const sceneName = await this.getCurrentSceneName();
    const response = await this.request('GetSceneItemList', { sceneName });
    const item = response.sceneItems.find((item: any) => item.sourceName === sourceName);
    return item ? item.sceneItemId : 0;
  }
}

// OBS Scene Manager
class OBSSceneManager {
  private client: OBSWebSocketClient;
  private scenes: OBSScene[] = [];
  private currentScene: string | null = null;
  
  constructor(client: OBSWebSocketClient) {
    this.client = client;
    
    // Listen for scene changes
    this.client.on('sceneChanged', (sceneName: string) => {
      this.currentScene = sceneName;
      this.updateSceneStatus();
    });
  }
  
  async initialize(): Promise<void> {
    try {
      this.scenes = await this.client.getSceneList();
      this.currentScene = this.scenes.find(s => s.isActive)?.name || null;
      console.log(`Loaded ${this.scenes.length} OBS scenes`);
    } catch (error) {
      console.error('Failed to initialize OBS scenes:', error);
      throw error;
    }
  }
  
  async switchToScene(sceneName: string): Promise<void> {
    try {
      await this.client.setCurrentScene(sceneName);
      this.currentScene = sceneName;
      this.updateSceneStatus();
      console.log(`Switched to OBS scene: ${sceneName}`);
    } catch (error) {
      console.error(`Failed to switch to scene ${sceneName}:`, error);
      throw error;
    }
  }
  
  getScenes(): OBSScene[] {
    return this.scenes;
  }
  
  getCurrentScene(): string | null {
    return this.currentScene;
  }
  
  private updateSceneStatus(): void {
    this.scenes.forEach(scene => {
      scene.isActive = scene.name === this.currentScene;
    });
  }
  
  async refreshScenes(): Promise<void> {
    this.scenes = await this.client.getSceneList();
  }
}

// OBS Automation Engine
class OBSAutomationEngine {
  private client: OBSWebSocketClient;
  private rules: OBSAutomationRule[] = [];
  private isEnabled = true;
  
  constructor(client: OBSWebSocketClient) {
    this.client = client;
  }
  
  addRule(rule: OBSAutomationRule): void {
    this.rules.push(rule);
    console.log(`Added OBS automation rule: ${rule.name}`);
  }
  
  removeRule(ruleId: string): void {
    this.rules = this.rules.filter(rule => rule.id !== ruleId);
  }
  
  async triggerRule(trigger: { type: string; [key: string]: any }): Promise<void> {
    if (!this.isEnabled) return;
    
    const matchingRules = this.rules.filter(rule => 
      rule.isEnabled && this.matchesTrigger(rule.trigger, trigger)
    );
    
    for (const rule of matchingRules) {
      try {
        if (await this.evaluateConditions(rule.conditions)) {
          await this.executeActions(rule.actions);
          rule.lastTriggered = new Date();
          console.log(`Executed OBS automation rule: ${rule.name}`);
        }
      } catch (error) {
        console.error(`Failed to execute OBS rule ${rule.name}:`, error);
      }
    }
  }
  
  private matchesTrigger(ruleTrigger: any, actualTrigger: any): boolean {
    if (ruleTrigger.type !== actualTrigger.type) return false;
    
    switch (ruleTrigger.type) {
      case 'button_press':
        return ruleTrigger.buttonId === actualTrigger.buttonId;
      case 'scene_change':
        return ruleTrigger.sceneId === actualTrigger.sceneId;
      default:
        return true;
    }
  }
  
  private async evaluateConditions(conditions: OBSCondition[]): Promise<boolean> {
    for (const condition of conditions) {
      if (!await this.evaluateCondition(condition)) {
        return false;
      }
    }
    return true;
  }
  
  private async evaluateCondition(condition: OBSCondition): Promise<boolean> {
    switch (condition.type) {
      case 'streaming':
        const streamStatus = await this.client.getStreamStatus();
        return streamStatus.isStreaming === condition.value;
      case 'scene_active':
        const currentScene = await this.client.request('GetCurrentProgramScene');
        return currentScene.currentProgramSceneName === condition.value;
      default:
        return true;
    }
  }
  
  private async executeActions(actions: OBSAction[]): Promise<void> {
    for (const action of actions) {
      try {
        if (action.delay) {
          await new Promise(resolve => setTimeout(resolve, action.delay));
        }
        
        await this.executeAction(action);
      } catch (error) {
        console.error(`Failed to execute OBS action ${action.type}:`, error);
      }
    }
  }
  
  private async executeAction(action: OBSAction): Promise<void> {
    switch (action.type) {
      case 'scene_switch':
        await this.client.setCurrentScene(action.targetId);
        break;
      case 'stream_start':
        await this.client.startStream();
        break;
      case 'stream_stop':
        await this.client.stopStream();
        break;
      case 'recording_start':
        await this.client.startRecording();
        break;
      case 'recording_stop':
        await this.client.stopRecording();
        break;
      case 'source_toggle':
        const visible = action.parameters.visible ?? true;
        await this.client.setSourceVisibility(action.targetId, visible);
        break;
      case 'source_volume':
        const volume = action.parameters.volume ?? 1.0;
        await this.client.setSourceVolume(action.targetId, volume);
        break;
      default:
        console.warn(`Unknown OBS action type: ${action.type}`);
    }
  }
  
  setEnabled(enabled: boolean): void {
    this.isEnabled = enabled;
    console.log(`OBS automation ${enabled ? 'enabled' : 'disabled'}`);
  }
  
  getRules(): OBSAutomationRule[] {
    return this.rules;
  }
}
```
