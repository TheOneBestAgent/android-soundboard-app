# Story 5.1: TLS & Pinning Implementation

## Status
Draft

## Story
**As a** security-conscious content creator using the soundboard app,
**I want** all communications between my mobile device and desktop/server to be secured with TLS 1.3 and certificate pinning,
**so that** my audio content, pairing sessions, and personal data are protected from man-in-the-middle attacks and unauthorized access.

## Acceptance Criteria
1. All WebSocket connections use TLS 1.3 encryption with perfect forward secrecy
2. Certificate pinning implemented for mobile app connections to prevent MITM attacks
3. mTLS (mutual TLS) authentication for device pairing and verification
4. Automatic certificate rotation and pinning updates without app store releases
5. Secure certificate storage using platform-native secure storage (Keychain/Keystore)
6. Certificate validation with proper error handling and user notifications
7. Fallback mechanisms for certificate pinning failures with user consent
8. HMAC-signed REST API calls for additional authentication layer
9. Rate limiting on security-sensitive endpoints (max 5 attempts per minute)
10. Comprehensive security logging for audit trails and monitoring
11. Compliance with OWASP Top 10 security guidelines
12. Performance impact <50ms additional latency for TLS handshake

### TLS Manager Implementation
```typescript
// Comprehensive TLS management for React Native
class TLSManager {
  private config: TLSConfiguration;
  private certificatePinning: CertificatePinning;
  private secureStorage: SecureStorage;
  private eventLogger: SecurityEventLogger;
  
  constructor() {
    this.secureStorage = new SecureStorage();
    this.eventLogger = new SecurityEventLogger();
    this.certificatePinning = new CertificatePinning(this.eventLogger);
  }
  
  async initialize(): Promise<void> {
    try {
      // Load TLS configuration
      this.config = await this.loadTLSConfiguration();
      
      // Initialize certificate pinning
      await this.certificatePinning.initialize(this.config.certificatePinning);
      
      // Setup WebSocket with TLS
      await this.setupSecureWebSocket();
      
      // Initialize mTLS if enabled
      if (this.config.mTLS.enabled) {
        await this.initializeMTLS();
      }
      
      console.log('TLS Manager initialized successfully');
    } catch (error) {
      await this.eventLogger.logSecurityEvent({
        type: 'tls_initialization_error',
        severity: 'critical',
        details: {
          errorMessage: error.message,
          timestamp: new Date(),
        },
      });
      throw error;
    }
  }
  
  private async setupSecureWebSocket(): Promise<void> {
    const wsConfig = {
      // TLS 1.3 configuration
      secureProtocol: 'TLSv1_3_method',
      ciphers: [
        'TLS_AES_256_GCM_SHA384',
        'TLS_CHACHA20_POLY1305_SHA256',
        'TLS_AES_128_GCM_SHA256',
      ].join(':'),
      
      // Certificate pinning
      checkServerIdentity: (hostname: string, cert: any) => {
        return this.certificatePinning.validateCertificate(hostname, cert);
      },
      
      // Client certificate for mTLS
      cert: this.config.mTLS.enabled ? await this.getClientCertificate() : undefined,
      key: this.config.mTLS.enabled ? await this.getClientPrivateKey() : undefined,
      
      // Additional security options
      rejectUnauthorized: true,
      requestCert: this.config.mTLS.enabled,
      agent: false,
    };
    
    // Configure WebSocket with TLS options
    global.WebSocket = class SecureWebSocket extends WebSocket {
      constructor(url: string, protocols?: string | string[]) {
        super(url, protocols, {
          headers: {
            'User-Agent': 'SoundboardApp/1.0',
            'X-Client-Version': '1.0.0',
          },
          ...wsConfig,
        });
      }
    };
  }
  
  private async initializeMTLS(): Promise<void> {
    try {
      // Check if client certificate exists
      let clientCert = await this.secureStorage.getClientCertificate();
      
      if (!clientCert || this.isCertificateExpiring(clientCert)) {
        // Generate new client certificate
        clientCert = await this.generateClientCertificate();
        await this.secureStorage.storeClientCertificate(clientCert);
      }
      
      // Validate certificate chain
      await this.validateCertificateChain(clientCert);
      
      console.log('mTLS initialized successfully');
    } catch (error) {
      await this.eventLogger.logSecurityEvent({
        type: 'mtls_initialization_error',
        severity: 'high',
        details: {
          errorMessage: error.message,
          timestamp: new Date(),
        },
      });
      throw error;
    }
  }
  
  private async generateClientCertificate(): Promise<ClientCertificate> {
    const deviceId = await this.getDeviceId();
    
    // Generate key pair
    const keyPair = await this.generateKeyPair();
    
    // Create certificate signing request
    const csr = await this.createCSR(keyPair, deviceId);
    
    // Send CSR to CA for signing
    const signedCert = await this.requestCertificateSigning(csr);
    
    return {
      id: generateId(),
      deviceId,
      certificate: signedCert.certificate,
      privateKey: await this.encryptPrivateKey(keyPair.privateKey),
      issuer: signedCert.issuer,
      subject: signedCert.subject,
      serialNumber: signedCert.serialNumber,
      validFrom: signedCert.validFrom,
      validTo: signedCert.validTo,
      fingerprint: await this.calculateFingerprint(signedCert.certificate),
      isRevoked: false,
      createdAt: new Date(),
      lastUsed: new Date(),
    };
  }
  
  async rotateCertificates(): Promise<void> {
    try {
      console.log('Starting certificate rotation...');
      
      // Rotate certificate pins
      await this.certificatePinning.rotatePins();
      
      // Rotate client certificate if needed
      if (this.config.mTLS.enabled) {
        const currentCert = await this.secureStorage.getClientCertificate();
        if (this.isCertificateExpiring(currentCert, 30)) { // 30 days before expiry
          await this.renewClientCertificate();
        }
      }
      
      await this.eventLogger.logSecurityEvent({
        type: 'certificate_rotation',
        severity: 'low',
        details: {
          timestamp: new Date(),
        },
      });
      
      console.log('Certificate rotation completed successfully');
    } catch (error) {
      await this.eventLogger.logSecurityEvent({
        type: 'certificate_rotation_error',
        severity: 'high',
        details: {
          errorMessage: error.message,
          timestamp: new Date(),
        },
      });
      throw error;
    }
  }
  
  private isCertificateExpiring(cert: ClientCertificate, daysThreshold = 7): boolean {
    const expiryDate = new Date(cert.validTo);
    const thresholdDate = new Date();
    thresholdDate.setDate(thresholdDate.getDate() + daysThreshold);
    
    return expiryDate <= thresholdDate;
  }
}

// Certificate pinning implementation
class CertificatePinning {
  private pins: Map<string, CertificatePin[]> = new Map();
  private eventLogger: SecurityEventLogger;
  
  constructor(eventLogger: SecurityEventLogger) {
    this.eventLogger = eventLogger;
  }
  
  async initialize(config: TLSConfiguration['certificatePinning']): Promise<void> {
    if (!config.enabled) {
      console.log('Certificate pinning disabled');
      return;
    }
    
    // Load certificate pins
    for (const pin of config.pins) {
      const hostname = pin.hostname;
      if (!this.pins.has(hostname)) {
        this.pins.set(hostname, []);
      }
      this.pins.get(hostname)!.push(pin);
    }
    
    console.log(`Certificate pinning initialized for ${this.pins.size} hostnames`);
  }
  
  async validateCertificate(hostname: string, cert: any): Promise<boolean> {
    const pins = this.pins.get(hostname);
    if (!pins || pins.length === 0) {
      // No pins configured for this hostname
      return true;
    }
    
    try {
      // Calculate certificate fingerprint
      const certFingerprint = await this.calculateCertificateFingerprint(cert);
      
      // Check against pinned certificates
      const isValid = pins.some(pin => {
        return pin.digest === certFingerprint && new Date() < pin.expiresAt;
      });
      
      if (!isValid) {
        await this.eventLogger.logSecurityEvent({
          type: 'certificate_pinning_failure',
          severity: 'critical',
          details: {
            hostname,
            actualPin: certFingerprint,
            expectedPin: pins.map(p => p.digest).join(', '),
            timestamp: new Date(),
          },
        });
        
        // In strict mode, reject the connection
        // In permissive mode, log but allow connection
        return false;
      }
      
      return true;
    } catch (error) {
      await this.eventLogger.logSecurityEvent({
        type: 'certificate_validation_error',
        severity: 'high',
        details: {
          hostname,
          errorMessage: error.message,
          timestamp: new Date(),
        },
      });
      
      return false;
    }
  }
  
  async rotatePins(): Promise<void> {
    // Fetch new certificate pins from server
    const newPins = await this.fetchUpdatedPins();
    
    // Update local pin storage
    for (const pin of newPins) {
      const hostname = pin.hostname;
      if (!this.pins.has(hostname)) {
        this.pins.set(hostname, []);
      }
      
      // Remove expired pins
      const currentPins = this.pins.get(hostname)!;
      const validPins = currentPins.filter(p => new Date() < p.expiresAt);
      
      // Add new pin
      validPins.push(pin);
      
      this.pins.set(hostname, validPins);
    }
    
    console.log('Certificate pins rotated successfully');
  }
  
  private async calculateCertificateFingerprint(cert: any): Promise<string> {
    // Implementation depends on platform
    // For React Native, use native modules
    if (Platform.OS === 'android') {
      return await AndroidCertificateModule.calculateFingerprint(cert);
    } else {
      return await IOSCertificateModule.calculateFingerprint(cert);
    }
  }
}

// HMAC authentication for API calls
class HMACAuth {
  private secretKey: string;
  private algorithm = 'sha256';
  
  constructor(secretKey: string) {
    this.secretKey = secretKey;
  }
  
  async signRequest(request: {
    method: string;
    url: string;
    headers: Record<string, string>;
    body?: string;
  }): Promise<HMACRequest> {
    const timestamp = Date.now();
    const nonce = await this.generateNonce();
    
    // Create signature payload
    const payload = [
      request.method.toUpperCase(),
      request.url,
      JSON.stringify(request.headers),
      request.body || '',
      timestamp.toString(),
      nonce,
    ].join('\n');
    
    // Generate HMAC signature
    const signature = await this.generateHMAC(payload);
    
    return {
      ...request,
      timestamp,
      nonce,
      signature,
    };
  }
  
  async verifyRequest(request: HMACRequest): Promise<boolean> {
    // Check timestamp (prevent replay attacks)
    const now = Date.now();
    const requestAge = now - request.timestamp;
    if (requestAge > 300000) { // 5 minutes
      return false;
    }
    
    // Recreate signature
    const payload = [
      request.method.toUpperCase(),
      request.url,
      JSON.stringify(request.headers),
      request.body || '',
      request.timestamp.toString(),
      request.nonce,
    ].join('\n');
    
    const expectedSignature = await this.generateHMAC(payload);
    
    // Constant-time comparison to prevent timing attacks
    return this.constantTimeCompare(request.signature, expectedSignature);
  }
  
  private async generateHMAC(payload: string): Promise<string> {
    const encoder = new TextEncoder();
    const keyData = encoder.encode(this.secretKey);
    const payloadData = encoder.encode(payload);
    
    const cryptoKey = await crypto.subtle.importKey(
      'raw',
      keyData,
      { name: 'HMAC', hash: 'SHA-256' },
      false,
      ['sign']
    );
    
    const signature = await crypto.subtle.sign('HMAC', cryptoKey, payloadData);
    
    return Array.from(new Uint8Array(signature))
      .map(b => b.toString(16).padStart(2, '0'))
      .join('');
  }
  
  private constantTimeCompare(a: string, b: string): boolean {
    if (a.length !== b.length) {
      return false;
    }
    
    let result = 0;
    for (let i = 0; i < a.length; i++) {
      result |= a.charCodeAt(i) ^ b.charCodeAt(i);
    }
    
    return result === 0;
  }
  
  private async generateNonce(): Promise<string> {
    const array = new Uint8Array(16);
    crypto.getRandomValues(array);
    return Array.from(array, byte => byte.toString(16).padStart(2, '0')).join('');
  }
}
```
