# Story 2.2: Grid Customization & Live Preview

## Status
Draft

## Story
**As a** content creator setting up my soundboard,
**I want** to customize the grid layout (4×3 to 10×10) with live preview of changes,
**so that** I can optimize my button arrangement for my streaming workflow with immediate visual feedback.

## Acceptance Criteria
1. Grid size adjustable from 4×3 minimum to 10×10 maximum
2. Live preview renders layout changes within ≤100ms
3. Drag-and-drop button repositioning with visual feedback
4. Real-time button customization (colors, labels, icons)
5. Layout presets for common configurations (4×3, 6×4, 8×6, 10×10)
6. Memory footprint stays under 200MB during grid editing
7. Undo/redo functionality for layout changes (up to 20 steps)
8. Auto-save layout changes every 30 seconds

### Grid Layout Calculations
```typescript
// Grid layout mathematics
class GridLayoutManager {
  calculateCellSize(containerWidth: number, containerHeight: number, rows: number, cols: number): { width: number; height: number } {
    const padding = 16;
    const spacing = 8;
    
    const availableWidth = containerWidth - (padding * 2) - (spacing * (cols - 1));
    const availableHeight = containerHeight - (padding * 2) - (spacing * (rows - 1));
    
    const cellWidth = Math.floor(availableWidth / cols);
    const cellHeight = Math.floor(availableHeight / rows);
    
    // Maintain aspect ratio for buttons
    const size = Math.min(cellWidth, cellHeight);
    return { width: size, height: size };
  }
  
  getGridPosition(x: number, y: number, cellSize: { width: number; height: number }, spacing: number): GridPosition {
    const col = Math.floor(x / (cellSize.width + spacing));
    const row = Math.floor(y / (cellSize.height + spacing));
    return { row, col };
  }
  
  validateLayout(layout: SoundboardLayout): boolean {
    return layout.rows >= 3 && layout.rows <= 10 && 
           layout.columns >= 3 && layout.columns <= 10;
  }
}
```

### Drag & Drop Implementation
```typescript
// React Native drag and drop with Gesture Handler
import { PanGestureHandler, State } from 'react-native-gesture-handler';
import Animated, { useSharedValue, useAnimatedStyle, runOnJS } from 'react-native-reanimated';

const DraggableButton: React.FC<DraggableButtonProps> = ({ button, onDrop }) => {
  const translateX = useSharedValue(0);
  const translateY = useSharedValue(0);
  const scale = useSharedValue(1);
  
  const gestureHandler = useAnimatedGestureHandler({
    onStart: () => {
      scale.value = withSpring(1.1);
    },
    onActive: (event) => {
      translateX.value = event.translationX;
      translateY.value = event.translationY;
    },
    onEnd: (event) => {
      scale.value = withSpring(1);
      translateX.value = withSpring(0);
      translateY.value = withSpring(0);
      
      runOnJS(handleDrop)(event.absoluteX, event.absoluteY);
    },
  });
  
  const animatedStyle = useAnimatedStyle(() => ({
    transform: [
      { translateX: translateX.value },
      { translateY: translateY.value },
      { scale: scale.value },
    ],
  }));
  
  return (
    <PanGestureHandler onGestureEvent={gestureHandler}>
      <Animated.View style={animatedStyle}>
        <SoundButton button={button} />
      </Animated.View>
    </PanGestureHandler>
  );
};
```

### Live Preview System
```typescript
// Preview rendering with performance optimization
class PreviewRenderer {
  private renderQueue: (() => void)[] = [];
  private isRendering = false;
  private lastRenderTime = 0;
  private readonly RENDER_THROTTLE = 16; // ~60fps
  
  queuePreviewUpdate(updateFn: () => void): void {
    this.renderQueue.push(updateFn);
    this.scheduleRender();
  }
  
  private scheduleRender(): void {
    if (this.isRendering) return;
    
    const now = Date.now();
    const timeSinceLastRender = now - this.lastRenderTime;
    
    if (timeSinceLastRender >= this.RENDER_THROTTLE) {
      this.executeRender();
    } else {
      setTimeout(() => this.executeRender(), this.RENDER_THROTTLE - timeSinceLastRender);
    }
  }
  
  private executeRender(): void {
    this.isRendering = true;
    this.lastRenderTime = Date.now();
    
    // Batch all queued updates
    const updates = [...this.renderQueue];
    this.renderQueue.length = 0;
    
    updates.forEach(update => update());
    
    this.isRendering = false;
    
    // Process any updates that came in during rendering
    if (this.renderQueue.length > 0) {
      this.scheduleRender();
    }
  }
}
```
