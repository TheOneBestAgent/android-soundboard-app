# Story 6.1: Smart Button Customization & Album Art

## Status
Draft

## Story
**As a** content creator setting up my soundboard,
**I want** automatic album art detection and smart button customization based on audio metadata,
**so that** my soundboard looks professional and visually appealing without manual customization effort.

## Acceptance Criteria
1. Automatic album art detection from audio file metadata (ID3 tags)
2. Online album art fetching from MusicBrainz/Last.fm APIs with <2s load time
3. Smart fallback system: metadata → online search → custom image → text → emoji → gradient
4. Adaptive color theming based on album art dominant colors
5. Automatic button labeling from audio metadata (title, artist, album)
6. Haptic feedback animations synchronized with button press
7. Real-time preview of customization changes (<100ms)
8. Batch auto-customization for multiple buttons simultaneously
9. Manual override options for all automatic customizations
10. Caching system for fetched album art and metadata
11. Accessibility support with alt-text and high contrast options
12. Integration with system theme (Material You on Android, iOS design language)

```typescript
// Audio metadata extraction and processing
class MetadataExtractionService {
  private metadataCache: Map<string, AudioMetadata> = new Map();
  private extractionQueue: string[] = [];
  private isProcessing = false;
  
  async extractMetadata(filePath: string): Promise<AudioMetadata> {
    // Check cache first
    const cached = this.metadataCache.get(filePath);
    if (cached && this.isCacheValid(cached)) {
      return cached;
    }
    
    try {
      // Use native module for efficient metadata extraction
      const rawMetadata = await ID3TagExtractorModule.extractMetadata(filePath);
      
      // Process and clean metadata
      const metadata: AudioMetadata = {
        id: generateId(),
        filePath,
        title: this.cleanText(rawMetadata.title),
        artist: this.cleanText(rawMetadata.artist),
        album: this.cleanText(rawMetadata.album),
        albumArtist: this.cleanText(rawMetadata.albumArtist),
        year: this.parseYear(rawMetadata.year),
        genre: this.normalizeGenre(rawMetadata.genre),
        track: this.parseTrackNumber(rawMetadata.track),
        duration: rawMetadata.duration || 0,
        bitrate: rawMetadata.bitrate,
        sampleRate: rawMetadata.sampleRate,
        embeddedArt: rawMetadata.embeddedArt ? {
          data: rawMetadata.embeddedArt.data,
          format: rawMetadata.embeddedArt.format,
          width: rawMetadata.embeddedArt.width,
          height: rawMetadata.embeddedArt.height,
        } : undefined,
        extractedAt: new Date(),
        source: 'id3',
      };
      
      // Apply fallback metadata if missing
      if (!metadata.title || !metadata.artist) {
        const fallbackMetadata = this.generateFallbackMetadata(filePath);
        metadata.title = metadata.title || fallbackMetadata.title;
        metadata.artist = metadata.artist || fallbackMetadata.artist;
        metadata.source = 'filename';
      }
      
      // Cache the result
      this.metadataCache.set(filePath, metadata);
      
      return metadata;
    } catch (error) {
      console.error('Metadata extraction failed:', error);
      
      // Return minimal metadata from filename
      return this.generateFallbackMetadata(filePath);
    }
  }
  
  private generateFallbackMetadata(filePath: string): AudioMetadata {
    const filename = path.basename(filePath, path.extname(filePath));
    
    // Try to parse "Artist - Title" format
    const parts = filename.split(' - ');
    let title = filename;
    let artist = 'Unknown Artist';
    
    if (parts.length >= 2) {
      artist = parts[0].trim();
      title = parts.slice(1).join(' - ').trim();
    }
    
    return {
      id: generateId(),
      filePath,
      title,
      artist,
      duration: 0,
      extractedAt: new Date(),
      source: 'filename',
    };
  }
  
  private cleanText(text?: string): string | undefined {
    if (!text) return undefined;
    
    return text
      .trim()
      .replace(/\0/g, '') // Remove null bytes
      .replace(/[\x00-\x1F\x7F]/g, '') // Remove control characters
      .substring(0, 200); // Limit length
  }
  
  private normalizeGenre(genre?: string): string | undefined {
    if (!genre) return undefined;
    
    // Normalize common genre variations
    const genreMap: Record<string, string> = {
      'rock': 'Rock',
      'pop': 'Pop',
      'hip-hop': 'Hip-Hop',
      'hiphop': 'Hip-Hop',
      'rap': 'Hip-Hop',
      'electronic': 'Electronic',
      'edm': 'Electronic',
      'classical': 'Classical',
      'jazz': 'Jazz',
      'blues': 'Blues',
      'country': 'Country',
      'folk': 'Folk',
      'metal': 'Metal',
      'punk': 'Punk',
      'reggae': 'Reggae',
      'r&b': 'R&B',
      'rnb': 'R&B',
      'soul': 'Soul',
      'funk': 'Funk',
    };
    
    const normalized = genre.toLowerCase().trim();
    return genreMap[normalized] || this.capitalizeWords(genre);
  }
  
  async batchExtractMetadata(filePaths: string[]): Promise<Map<string, AudioMetadata>> {
    const results = new Map<string, AudioMetadata>();
    
    // Process in batches to avoid overwhelming the system
    const batchSize = 5;
    for (let i = 0; i < filePaths.length; i += batchSize) {
      const batch = filePaths.slice(i, i + batchSize);
      
      const batchPromises = batch.map(async (filePath) => {
        try {
          const metadata = await this.extractMetadata(filePath);
          results.set(filePath, metadata);
        } catch (error) {
          console.error(`Failed to extract metadata for ${filePath}:`, error);
        }
      });
      
      await Promise.all(batchPromises);
      
      // Small delay between batches
      if (i + batchSize < filePaths.length) {
        await new Promise(resolve => setTimeout(resolve, 100));
      }
    }
    
    return results;
  }
}
```

### Album Art Fetching System
```typescript
// Album art fetching from multiple sources
class AlbumArtService {
  private musicBrainzService: MusicBrainzService;
  private lastFmService: LastFmService;
  private artCache: Map<string, AlbumArtResult> = new Map();
  private fetchQueue: Map<string, Promise<AlbumArtResult | null>> = new Map();
  
  constructor() {
    this.musicBrainzService = new MusicBrainzService();
    this.lastFmService = new LastFmService();
  }
  
  async fetchAlbumArt(metadata: AudioMetadata): Promise<AlbumArtResult | null> {
    const cacheKey = this.generateCacheKey(metadata);
    
    // Check cache first
    const cached = this.artCache.get(cacheKey);
    if (cached && this.isCacheValid(cached)) {
      return cached;
    }
    
    // Check if already fetching
    const existingFetch = this.fetchQueue.get(cacheKey);
    if (existingFetch) {
      return existingFetch;
    }
    
    // Start new fetch
    const fetchPromise = this.performAlbumArtFetch(metadata, cacheKey);
    this.fetchQueue.set(cacheKey, fetchPromise);
    
    try {
      const result = await fetchPromise;
      return result;
    } finally {
      this.fetchQueue.delete(cacheKey);
    }
  }
  
  private async performAlbumArtFetch(
    metadata: AudioMetadata, 
    cacheKey: string
  ): Promise<AlbumArtResult | null> {
    // 1. Check for embedded album art first
    if (metadata.embeddedArt) {
      const embeddedResult = await this.processEmbeddedArt(metadata);
      if (embeddedResult) {
        this.artCache.set(cacheKey, embeddedResult);
        return embeddedResult;
      }
    }
    
    // 2. Try MusicBrainz API
    if (metadata.artist && metadata.album) {
      try {
        const mbResult = await this.musicBrainzService.searchAlbumArt(
          metadata.artist,
          metadata.album
        );
        
        if (mbResult) {
          const processedResult = await this.processOnlineArt(metadata, mbResult, 'musicbrainz');
          this.artCache.set(cacheKey, processedResult);
          return processedResult;
        }
      } catch (error) {
        console.warn('MusicBrainz search failed:', error);
      }
    }
    
    // 3. Try Last.fm API
    if (metadata.artist && (metadata.album || metadata.title)) {
      try {
        const lastFmResult = await this.lastFmService.searchAlbumArt(
          metadata.artist,
          metadata.album || metadata.title
        );
        
        if (lastFmResult) {
          const processedResult = await this.processOnlineArt(metadata, lastFmResult, 'lastfm');
          this.artCache.set(cacheKey, processedResult);
          return processedResult;
        }
      } catch (error) {
        console.warn('Last.fm search failed:', error);
      }
    }
    
    // 4. Generate fallback art
    const fallbackResult = await this.generateFallbackArt(metadata);
    this.artCache.set(cacheKey, fallbackResult);
    return fallbackResult;
  }
  
  private async processEmbeddedArt(metadata: AudioMetadata): Promise<AlbumArtResult | null> {
    if (!metadata.embeddedArt) return null;
    
    try {
      // Save embedded art to local storage
      const filename = `embedded_${metadata.id}.${metadata.embeddedArt.format}`;
      const localPath = `${RNFS.DocumentDirectoryPath}/album_art/${filename}`;
      
      await RNFS.writeFile(localPath, metadata.embeddedArt.data, 'base64');
      
      // Extract dominant colors
      const dominantColors = await ColorExtractionModule.extractColors(localPath);
      
      return {
        id: generateId(),
        audioId: metadata.id,
        localPath,
        source: 'embedded',
        quality: this.determineQuality(metadata.embeddedArt.width, metadata.embeddedArt.height),
        dimensions: {
          width: metadata.embeddedArt.width,
          height: metadata.embeddedArt.height,
        },
        dominantColors,
        confidence: 1.0,
        fetchedAt: new Date(),
      };
    } catch (error) {
      console.error('Failed to process embedded art:', error);
      return null;
    }
  }
  
  private async processOnlineArt(
    metadata: AudioMetadata,
    artUrl: string,
    source: 'musicbrainz' | 'lastfm'
  ): Promise<AlbumArtResult> {
    // Download and cache the image
    const filename = `${source}_${metadata.id}.jpg`;
    const localPath = `${RNFS.DocumentDirectoryPath}/album_art/${filename}`;
    
    const downloadResult = await RNFS.downloadFile({
      fromUrl: artUrl,
      toFile: localPath,
    }).promise;
    
    if (downloadResult.statusCode !== 200) {
      throw new Error(`Failed to download album art: ${downloadResult.statusCode}`);
    }
    
    // Get image dimensions
    const dimensions = await this.getImageDimensions(localPath);
    
    // Extract dominant colors
    const dominantColors = await ColorExtractionModule.extractColors(localPath);
    
    return {
      id: generateId(),
      audioId: metadata.id,
      imageUrl: artUrl,
      localPath,
      source,
      quality: this.determineQuality(dimensions.width, dimensions.height),
      dimensions,
      dominantColors,
      confidence: source === 'musicbrainz' ? 0.9 : 0.7,
      fetchedAt: new Date(),
      expiresAt: new Date(Date.now() + 30 * 24 * 60 * 60 * 1000), // 30 days
    };
  }
  
  private async generateFallbackArt(metadata: AudioMetadata): Promise<AlbumArtResult> {
    // Generate a gradient or pattern based on metadata
    const colors = this.generateColorsFromMetadata(metadata);
    
    const fallbackArt = await this.createGradientImage(colors, {
      width: 512,
      height: 512,
    });
    
    const filename = `fallback_${metadata.id}.png`;
    const localPath = `${RNFS.DocumentDirectoryPath}/album_art/${filename}`;
    
    await RNFS.writeFile(localPath, fallbackArt, 'base64');
    
    return {
      id: generateId(),
      audioId: metadata.id,
      localPath,
      source: 'generated',
      quality: 'high',
      dimensions: { width: 512, height: 512 },
      dominantColors: {
        primary: colors[0],
        secondary: colors[1],
        accent: colors[2] || colors[0],
        background: colors[0],
        text: this.getContrastColor(colors[0]),
      },
      confidence: 0.3,
      fetchedAt: new Date(),
    };
  }
  
  private generateColorsFromMetadata(metadata: AudioMetadata): string[] {
    // Generate colors based on genre, artist name, etc.
    const genreColors: Record<string, string[]> = {
      'Rock': ['#FF6B6B', '#4ECDC4', '#45B7D1'],
      'Pop': ['#FF9FF3', '#54A0FF', '#5F27CD'],
      'Hip-Hop': ['#FFA502', '#FF6348', '#2F3542'],
      'Electronic': ['#3742FA', '#2F3542', '#FF3838'],
      'Classical': ['#A4B0BE', '#57606F', '#2F3542'],
      'Jazz': ['#FFA502', '#FF6348', '#2F3542'],
      'Blues': ['#3742FA', '#40739E', '#487EB0'],
      'Country': ['#7BED9F', '#70A1FF', '#5352ED'],
      'Metal': ['#2F3542', '#57606F', '#A4B0BE'],
      'Punk': ['#FF3838', '#FF6B6B', '#2F3542'],
    };
    
    const genre = metadata.genre || 'Unknown';
    const colors = genreColors[genre] || ['#667EEA', '#764BA2', '#F093FB'];
    
    // Add some randomness based on artist name
    if (metadata.artist) {
      const hash = this.simpleHash(metadata.artist);
      const hue = (hash % 360);
      colors.push(`hsl(${hue}, 70%, 60%)`);
    }
    
    return colors;
  }
}
```

### Smart Customization Engine
```typescript
// Intelligent button customization based on metadata and album art
class SmartCustomizationService {
  private albumArtService: AlbumArtService;
  private colorAnalysisService: ColorAnalysisService;
  private templateStore: CustomizationTemplateStore;
  
  constructor() {
    this.albumArtService = new AlbumArtService();
    this.colorAnalysisService = new ColorAnalysisService();
    this.templateStore = new CustomizationTemplateStore();
  }
  
  async generateSmartCustomization(
    buttonId: string,
    audioMetadata: AudioMetadata
  ): Promise<SmartCustomization> {
    // 1. Fetch album art
    const albumArt = await this.albumArtService.fetchAlbumArt(audioMetadata);
    
    // 2. Find matching template
    const template = await this.findBestTemplate(audioMetadata, albumArt);
    
    // 3. Generate base customization
    const baseCustomization = this.generateBaseCustomization(audioMetadata, albumArt, template);
    
    // 4. Apply smart enhancements
    const enhancedCustomization = await this.applySmartEnhancements(
      baseCustomization,
      audioMetadata,
      albumArt
    );
    
    // 5. Calculate confidence score
    const confidence = this.calculateConfidence(audioMetadata, albumArt, template);
    
    const smartCustomization: SmartCustomization = {
      id: generateId(),
      buttonId,
      audioId: audioMetadata.id,
      isAutoGenerated: true,
      confidence,
      customization: enhancedCustomization,
      overrides: {
        background: false,
        label: false,
        icon: false,
        colors: false,
      },
      generatedAt: new Date(),
      lastModified: new Date(),
    };
    
    return smartCustomization;
  }
  
  private generateBaseCustomization(
    metadata: AudioMetadata,
    albumArt: AlbumArtResult | null,
    template: CustomizationTemplate | null
  ): SmartCustomization['customization'] {
    const colors = albumArt?.dominantColors || this.generateFallbackColors(metadata);
    
    return {
      background: {
        type: albumArt ? 'image' : 'gradient',
        source: albumArt?.localPath,
        colors: albumArt ? [colors.background] : [colors.primary, colors.secondary],
        opacity: 1,
      },
      label: {
        text: this.generateLabelText(metadata),
        color: colors.text,
        font: {
          family: this.selectFont(metadata.genre),
          size: this.calculateFontSize(metadata.title || ''),
          weight: 'normal',
          style: 'normal',
        },
        position: albumArt ? 'bottom' : 'center',
        shadow: {
          enabled: true,
          color: 'rgba(0, 0, 0, 0.5)',
          offset: { x: 0, y: 1 },
          blur: 2,
        },
      },
      border: {
        width: 2,
        color: colors.accent,
        radius: 8,
      },
      animation: {
        press: this.selectAnimation(metadata.genre),
        duration: 200,
        haptic: 'medium',
      },
    };
  }
  
  private async applySmartEnhancements(
    baseCustomization: SmartCustomization['customization'],
    metadata: AudioMetadata,
    albumArt: AlbumArtResult | null
  ): Promise<SmartCustomization['customization']> {
    const enhanced = { ...baseCustomization };
    
    // Enhance based on audio characteristics
    if (metadata.genre) {
      enhanced.animation = this.enhanceAnimationForGenre(enhanced.animation, metadata.genre);
    }
    
    // Enhance contrast if needed
    if (albumArt) {
      const contrastRatio = this.colorAnalysisService.calculateContrast(
        enhanced.label.color,
        enhanced.background.colors[0]
      );
      
      if (contrastRatio < 4.5) {
        enhanced.label.color = this.colorAnalysisService.adjustForContrast(
          enhanced.label.color,
          enhanced.background.colors[0]
        );
        enhanced.label.shadow.enabled = true;
      }
    }
    
    // Add icon if appropriate
    const icon = this.selectIcon(metadata);
    if (icon) {
      enhanced.icon = {
        type: icon.type,
        source: icon.source,
        size: 24,
        color: enhanced.label.color,
        position: 'top-right',
      };
    }
    
    return enhanced;
  }
  
  private generateLabelText(metadata: AudioMetadata): string {
    // Prioritize title, fallback to filename
    if (metadata.title) {
      // Include artist if it fits
      if (metadata.artist && (metadata.title.length + metadata.artist.length) < 30) {
        return `${metadata.title}\n${metadata.artist}`;
      }
      return metadata.title;
    }
    
    // Extract from filename
    const filename = path.basename(metadata.filePath, path.extname(metadata.filePath));
    return filename.length > 20 ? filename.substring(0, 17) + '...' : filename;
  }
  
  private selectFont(genre?: string): string {
    const fontMap: Record<string, string> = {
      'Rock': 'System',
      'Metal': 'System',
      'Punk': 'System',
      'Classical': 'serif',
      'Jazz': 'serif',
      'Electronic': 'monospace',
      'Hip-Hop': 'System',
      'Pop': 'System',
    };
    
    return fontMap[genre || ''] || 'System';
  }
  
  private selectAnimation(genre?: string): 'scale' | 'glow' | 'ripple' | 'bounce' {
    const animationMap: Record<string, 'scale' | 'glow' | 'ripple' | 'bounce'> = {
      'Electronic': 'glow',
      'Hip-Hop': 'bounce',
      'Pop': 'ripple',
      'Rock': 'scale',
      'Metal': 'scale',
      'Classical': 'ripple',
      'Jazz': 'ripple',
    };
    
    return animationMap[genre || ''] || 'scale';
  }
  
  private selectIcon(metadata: AudioMetadata): { type: 'emoji' | 'built-in'; source: string } | null {
    if (!metadata.genre) return null;
    
    const iconMap: Record<string, { type: 'emoji' | 'built-in'; source: string }> = {
      'Rock': { type: 'emoji', source: '🎸' },
      'Pop': { type: 'emoji', source: '🎤' },
      'Hip-Hop': { type: 'emoji', source: '🎤' },
      'Electronic': { type: 'emoji', source: '🎛️' },
      'Classical': { type: 'emoji', source: '🎼' },
      'Jazz': { type: 'emoji', source: '🎷' },
      'Blues': { type: 'emoji', source: '🎸' },
      'Country': { type: 'emoji', source: '🤠' },
      'Metal': { type: 'emoji', source: '🤘' },
      'Punk': { type: 'emoji', source: '🤘' },
    };
    
    return iconMap[metadata.genre] || null;
  }
  
  async batchGenerateCustomizations(
    buttonAudioPairs: Array<{ buttonId: string; audioMetadata: AudioMetadata }>
  ): Promise<SmartCustomization[]> {
    const results: SmartCustomization[] = [];
    
    // Process in batches to avoid overwhelming the system
    const batchSize = 3;
    for (let i = 0; i < buttonAudioPairs.length; i += batchSize) {
      const batch = buttonAudioPairs.slice(i, i + batchSize);
      
      const batchPromises = batch.map(async ({ buttonId, audioMetadata }) => {
        try {
          const customization = await this.generateSmartCustomization(buttonId, audioMetadata);
          results.push(customization);
        } catch (error) {
          console.error(`Failed to generate customization for button ${buttonId}:`, error);
        }
      });
      
      await Promise.all(batchPromises);
      
      // Progress callback
      const progress = Math.min(100, ((i + batchSize) / buttonAudioPairs.length) * 100);
      this.notifyBatchProgress(progress);
    }
    
    return results;
  }
  
  private calculateConfidence(
    metadata: AudioMetadata,
    albumArt: AlbumArtResult | null,
    template: CustomizationTemplate | null
  ): number {
    let confidence = 0.5; // Base confidence
    
    // Boost confidence based on available data
    if (metadata.title) confidence += 0.1;
    if (metadata.artist) confidence += 0.1;
    if (metadata.album) confidence += 0.1;
    if (metadata.genre) confidence += 0.1;
    if (albumArt && albumArt.confidence > 0.7) confidence += 0.2;
    if (template) confidence += 0.1;
    
    return Math.min(1.0, confidence);
  }
}
```
