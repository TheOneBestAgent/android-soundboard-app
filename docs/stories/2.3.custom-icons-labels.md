# Story 2.3: Custom Icons & Labels

## Status
Draft

## Story
**As a** content creator customizing my soundboard,
**I want** to assign custom icons, labels, and visual themes to my sound buttons,
**so that** I can create a personalized and visually organized soundboard that matches my brand and workflow.

## Acceptance Criteria
1. Custom label editing with text formatting (font, size, color, style)
2. Icon selection from built-in library (500+ icons) and custom icon upload
3. Button color customization with theme integration
4. Icon and label positioning options (top, bottom, left, right, center)
5. Real-time preview of customization changes
6. Bulk customization for multiple buttons simultaneously
7. Custom icon upload supports PNG, SVG, and JPEG formats (max 2MB)
8. Label text supports emoji and special characters
9. Customization templates and presets for quick styling
10. Export/import customization themes for sharing

### Button Customization Engine
```typescript
// Real-time button customization with preview
class ButtonCustomizationEngine {
  private previewRenderer: PreviewRenderer;
  private customizationCache: Map<string, ButtonCustomization> = new Map();
  
  constructor() {
    this.previewRenderer = new PreviewRenderer();
  }
  
  async customizeButton(
    buttonId: string, 
    customization: Partial<ButtonCustomization>
  ): Promise<ButtonCustomization> {
    const existing = await this.getButtonCustomization(buttonId);
    const updated = this.mergeCustomization(existing, customization);
    
    // Validate customization
    this.validateCustomization(updated);
    
    // Update cache
    this.customizationCache.set(buttonId, updated);
    
    // Trigger preview update
    this.previewRenderer.updatePreview(buttonId, updated);
    
    // Persist to storage
    await this.saveCustomization(buttonId, updated);
    
    return updated;
  }
  
  async bulkCustomize(
    buttonIds: string[], 
    customization: Partial<ButtonCustomization>
  ): Promise<void> {
    const operations = buttonIds.map(buttonId => 
      this.customizeButton(buttonId, customization)
    );
    
    await Promise.all(operations);
    
    // Trigger bulk preview update
    this.previewRenderer.updateBulkPreview(buttonIds);
  }
  
  generateAutoCustomization(audioMetadata: AudioMetadata): Partial<ButtonCustomization> {
    const genre = audioMetadata.genre?.toLowerCase();
    const mood = this.detectMood(audioMetadata);
    
    // Smart customization based on audio content
    const baseCustomization: Partial<ButtonCustomization> = {
      label: {
        text: audioMetadata.title || 'Sound',
        color: this.getColorForGenre(genre),
        font: {
          family: this.getFontForMood(mood),
          size: 14,
          weight: 'normal',
          style: 'normal',
        },
        position: 'bottom',
        alignment: 'center',
        maxLines: 2,
        ellipsize: true,
      },
      icon: {
        type: 'built-in',
        source: this.getIconForGenre(genre),
        size: 32,
        position: 'center',
        opacity: 0.8,
      },
      background: {
        type: 'gradient',
        colors: this.getGradientForMood(mood),
        opacity: 1,
      },
    };
    
    return baseCustomization;
  }
  
  private validateCustomization(customization: ButtonCustomization): void {
    // Validate label text length
    if (customization.label.text.length > 50) {
      throw new Error('Label text too long (max 50 characters)');
    }
    
    // Validate color formats
    if (!this.isValidColor(customization.label.color)) {
      throw new Error('Invalid label color format');
    }
    
    // Validate icon size
    if (customization.icon.size < 16 || customization.icon.size > 128) {
      throw new Error('Icon size must be between 16 and 128 pixels');
    }
    
    // Validate accessibility contrast
    if (!this.hasGoodContrast(customization.label.color, customization.background.colors[0])) {
      console.warn('Poor color contrast detected - may affect accessibility');
    }
  }
  
  private hasGoodContrast(foreground: string, background: string): boolean {
    const contrast = this.calculateContrast(foreground, background);
    return contrast >= 4.5; // WCAG AA standard
  }
}
```

### Icon Management System
```typescript
// Icon library and custom icon management
class IconManagementService {
  private builtInIcons: Map<string, IconDefinition> = new Map();
  private customIcons: Map<string, CustomIcon> = new Map();
  private iconCache: Map<string, string> = new Map(); // icon ID -> base64 data
  
  async initializeBuiltInIcons(): Promise<void> {
    // Load built-in icon library
    const iconManifest = await import('../assets/icons/manifest.json');
    
    for (const category of iconManifest.categories) {
      for (const icon of category.icons) {
        this.builtInIcons.set(icon.id, {
          id: icon.id,
          name: icon.name,
          category: category.name,
          path: icon.path,
          tags: icon.tags,
          isBuiltIn: true,
        });
      }
    }
  }
  
  async uploadCustomIcon(file: File): Promise<CustomIcon> {
    // Validate file
    this.validateIconFile(file);
    
    // Process and optimize icon
    const processedIcon = await this.processIcon(file);
    
    // Generate unique ID and save
    const iconId = generateId();
    const iconPath = `custom_icons/${iconId}.${processedIcon.format}`;
    
    await RNFS.writeFile(iconPath, processedIcon.data, 'base64');
    
    const customIcon: CustomIcon = {
      id: iconId,
      name: file.name.replace(/\.[^/.]+$/, ''),
      originalName: file.name,
      filePath: iconPath,
      format: processedIcon.format,
      size: file.size,
      dimensions: processedIcon.dimensions,
      uploadDate: new Date(),
      usageCount: 0,
      tags: [],
      isPublic: false,
    };
    
    this.customIcons.set(iconId, customIcon);
    await this.saveCustomIconMetadata(customIcon);
    
    return customIcon;
  }
  
  async searchIcons(query: string, category?: string): Promise<IconDefinition[]> {
    const allIcons = [...this.builtInIcons.values(), ...this.customIcons.values()];
    
    return allIcons.filter(icon => {
      const matchesQuery = !query || 
        icon.name.toLowerCase().includes(query.toLowerCase()) ||
        icon.tags.some(tag => tag.toLowerCase().includes(query.toLowerCase()));
      
      const matchesCategory = !category || icon.category === category;
      
      return matchesQuery && matchesCategory;
    });
  }
  
  async getIconData(iconId: string): Promise<string> {
    // Check cache first
    if (this.iconCache.has(iconId)) {
      return this.iconCache.get(iconId)!;
    }
    
    // Load icon data
    const icon = this.builtInIcons.get(iconId) || this.customIcons.get(iconId);
    if (!icon) {
      throw new Error(`Icon not found: ${iconId}`);
    }
    
    let iconData: string;
    
    if (icon.isBuiltIn) {
      // Load from app bundle
      iconData = await this.loadBuiltInIcon(icon.path);
    } else {
      // Load custom icon
      iconData = await RNFS.readFile(icon.filePath, 'base64');
    }
    
    // Cache for future use
    this.iconCache.set(iconId, iconData);
    
    return iconData;
  }
  
  private validateIconFile(file: File): void {
    // Check file size (max 2MB)
    if (file.size > 2 * 1024 * 1024) {
      throw new Error('Icon file too large (max 2MB)');
    }
    
    // Check file format
    const allowedFormats = ['image/png', 'image/svg+xml', 'image/jpeg'];
    if (!allowedFormats.includes(file.type)) {
      throw new Error('Unsupported icon format (PNG, SVG, JPEG only)');
    }
  }
  
  private async processIcon(file: File): Promise<ProcessedIcon> {
    // For SVG files, optimize and validate
    if (file.type === 'image/svg+xml') {
      return this.processSVGIcon(file);
    }
    
    // For raster images, resize and optimize
    return this.processRasterIcon(file);
  }
}
```

```typescript
// Theme management and application
class ThemeService {
  private themes: Map<string, CustomizationTheme> = new Map();
  private currentTheme: CustomizationTheme | null = null;
  
  async loadBuiltInThemes(): Promise<void> {
    const themeFiles = ['default', 'dark', 'gaming', 'streaming', 'minimal'];
    
    for (const themeName of themeFiles) {
      const themeData = await import(`../assets/themes/${themeName}.json`);
      const theme: CustomizationTheme = {
        ...themeData,
        id: themeName,
        isBuiltIn: true,
        createdAt: new Date(),
        updatedAt: new Date(),
      };
      
      this.themes.set(themeName, theme);
    }
  }
  
  async applyTheme(themeId: string, buttonIds: string[]): Promise<void> {
    const theme = this.themes.get(themeId);
    if (!theme) {
      throw new Error(`Theme not found: ${themeId}`);
    }
    
    this.currentTheme = theme;
    
    // Apply theme to all specified buttons
    const customizationEngine = ButtonCustomizationEngine.getInstance();
    
    for (const buttonId of buttonIds) {
      const themeCustomization = this.generateThemeCustomization(theme);
      await customizationEngine.customizeButton(buttonId, themeCustomization);
    }
  }
  
  async createCustomTheme(
    name: string, 
    baseThemeId: string, 
    customizations: Partial<CustomizationTheme>
  ): Promise<CustomizationTheme> {
    const baseTheme = this.themes.get(baseThemeId);
    if (!baseTheme) {
      throw new Error(`Base theme not found: ${baseThemeId}`);
    }
    
    const customTheme: CustomizationTheme = {
      ...baseTheme,
      ...customizations,
      id: generateId(),
      name,
      isBuiltIn: false,
      createdAt: new Date(),
      updatedAt: new Date(),
    };
    
    this.themes.set(customTheme.id, customTheme);
    await this.saveCustomTheme(customTheme);
    
    return customTheme;
  }
  
  async exportTheme(themeId: string): Promise<string> {
    const theme = this.themes.get(themeId);
    if (!theme) {
      throw new Error(`Theme not found: ${themeId}`);
    }
    
    // Create exportable theme package
    const exportData = {
      theme,
      version: '1.0',
      exportDate: new Date().toISOString(),
      requiredAssets: this.getThemeAssets(theme),
    };
    
    return JSON.stringify(exportData, null, 2);
  }
  
  async importTheme(themeData: string): Promise<CustomizationTheme> {
    const importData = JSON.parse(themeData);
    
    // Validate theme data
    this.validateThemeData(importData.theme);
    
    // Import required assets
    await this.importThemeAssets(importData.requiredAssets);
    
    // Add theme to library
    const theme = {
      ...importData.theme,
      id: generateId(),
      isBuiltIn: false,
      createdAt: new Date(),
      updatedAt: new Date(),
    };
    
    this.themes.set(theme.id, theme);
    await this.saveCustomTheme(theme);
    
    return theme;
  }
}
```
