# Story 7.2: YouTube Audio Sampling Integration

## Status
Draft

## Story
**As a** content creator building my soundboard,
**I want** to search, preview, and import audio clips directly from YouTube videos,
**so that** I can quickly add popular sounds, music clips, and audio content to my soundboard without manual downloading and conversion.

## Acceptance Criteria
1. YouTube video search with keyword, channel, and duration filters
2. Audio preview playback before importing (30-second clips)
3. Audio extraction and conversion to app-compatible formats (MP3, WAV)
4. Copyright compliance with proper attribution and usage warnings
5. Import progress tracking with cancel/retry functionality
6. Automatic metadata extraction (title, description, thumbnail, duration)
7. Quality selection options (128kbps, 192kbps, 320kbps)
8. Batch import support for multiple videos/clips
9. Integration with existing soundboard grid and button assignment
10. Offline access to previously imported YouTube audio clips
11. Rate limiting compliance with YouTube API quotas
12. Error handling for unavailable, private, or restricted content

```typescript
// YouTube Data API v3 integration
class YouTubeAPIService {
  private apiKey: string;
  private baseURL = 'https://www.googleapis.com/youtube/v3';
  private quotaManager: QuotaManager;
  
  constructor(apiKey: string) {
    this.apiKey = apiKey;
    this.quotaManager = new QuotaManager(10000); // 10k units per day
  }
  
  async searchVideos(params: YouTubeSearchParams): Promise<YouTubeVideo[]> {
    // Check quota before making request
    await this.quotaManager.checkQuota(100); // Search costs 100 units
    
    const searchParams = new URLSearchParams({
      part: 'snippet',
      type: 'video',
      key: this.apiKey,
      q: params.query,
      maxResults: params.maxResults.toString(),
      order: params.order,
      videoDuration: params.duration,
      safeSearch: params.safeSearch,
      ...(params.channelId && { channelId: params.channelId }),
      ...(params.publishedAfter && { publishedAfter: params.publishedAfter.toISOString() }),
      ...(params.regionCode && { regionCode: params.regionCode }),
    });
    
    try {
      const response = await fetch(`${this.baseURL}/search?${searchParams}`);
      
      if (!response.ok) {
        throw new YouTubeAPIError(`Search failed: ${response.status}`, response.status);
      }
      
      const data = await response.json();
      
      // Get detailed video information
      const videoIds = data.items.map((item: any) => item.id.videoId).join(',');
      const detailedVideos = await this.getVideoDetails(videoIds);
      
      return detailedVideos;
    } catch (error) {
      console.error('YouTube search error:', error);
      throw error;
    }
  }
  
  async getVideoDetails(videoIds: string): Promise<YouTubeVideo[]> {
    await this.quotaManager.checkQuota(1); // Videos endpoint costs 1 unit per video
    
    const params = new URLSearchParams({
      part: 'snippet,contentDetails,statistics,status',
      id: videoIds,
      key: this.apiKey,
    });
    
    const response = await fetch(`${this.baseURL}/videos?${params}`);
    
    if (!response.ok) {
      throw new YouTubeAPIError(`Video details failed: ${response.status}`, response.status);
    }
    
    const data = await response.json();
    
    return data.items.map((item: any) => ({
      id: item.id,
      title: item.snippet.title,
      description: item.snippet.description,
      channelId: item.snippet.channelId,
      channelTitle: item.snippet.channelTitle,
      thumbnails: item.snippet.thumbnails,
      duration: item.contentDetails.duration,
      publishedAt: new Date(item.snippet.publishedAt),
      viewCount: parseInt(item.statistics.viewCount || '0'),
      likeCount: parseInt(item.statistics.likeCount || '0'),
      tags: item.snippet.tags || [],
      categoryId: item.snippet.categoryId,
      liveBroadcastContent: item.snippet.liveBroadcastContent,
      contentRating: item.contentDetails.contentRating || {},
      regionRestriction: item.contentDetails.regionRestriction,
    }));
  }
  
  async checkVideoAvailability(videoId: string): Promise<boolean> {
    try {
      const videos = await this.getVideoDetails(videoId);
      return videos.length > 0 && videos[0].liveBroadcastContent === 'none';
    } catch (error) {
      return false;
    }
  }
}

// Quota management for API rate limiting
class QuotaManager {
  private dailyQuota: number;
  private usedQuota: number = 0;
  private lastReset: Date = new Date();
  
  constructor(dailyQuota: number) {
    this.dailyQuota = dailyQuota;
    this.loadQuotaUsage();
  }
  
  async checkQuota(cost: number): Promise<void> {
    this.resetIfNewDay();
    
    if (this.usedQuota + cost > this.dailyQuota) {
      throw new QuotaExceededError('YouTube API quota exceeded for today');
    }
    
    this.usedQuota += cost;
    await this.saveQuotaUsage();
  }
  
  private resetIfNewDay(): void {
    const now = new Date();
    const lastResetDate = new Date(this.lastReset);
    
    if (now.getDate() !== lastResetDate.getDate() || 
        now.getMonth() !== lastResetDate.getMonth() || 
        now.getFullYear() !== lastResetDate.getFullYear()) {
      this.usedQuota = 0;
      this.lastReset = now;
    }
  }
}
```

### Audio Extraction Engine
```typescript
// Audio extraction using youtube-dl/yt-dlp
class AudioExtractionService {
  private extractionQueue: Map<string, YouTubeImportJob> = new Map();
  private maxConcurrentExtractions = 3;
  private activeExtractions = 0;
  
  async extractAudio(
    videoId: string, 
    options: {
      quality: '128' | '192' | '320';
      format: 'mp3' | 'wav' | 'm4a';
      startTime?: number;
      endTime?: number;
    }
  ): Promise<YouTubeImportJob> {
    const jobId = generateId();
    const job: YouTubeImportJob = {
      id: jobId,
      videoId,
      videoTitle: '', // Will be filled during extraction
      status: 'pending',
      progress: 0,
      quality: options.quality,
      format: options.format,
      startTime: options.startTime,
      endTime: options.endTime,
      createdAt: new Date(),
      updatedAt: new Date(),
      attribution: {
        videoTitle: '',
        channelName: '',
        videoUrl: `https://youtube.com/watch?v=${videoId}`,
        importDate: new Date(),
        usage: 'personal',
        copyrightNotice: 'Content imported from YouTube. Original copyright belongs to the content creator.',
      },
    };
    
    this.extractionQueue.set(jobId, job);
    
    // Start extraction if under concurrent limit
    if (this.activeExtractions < this.maxConcurrentExtractions) {
      this.processExtractionJob(jobId);
    }
    
    return job;
  }
  
  private async processExtractionJob(jobId: string): Promise<void> {
    const job = this.extractionQueue.get(jobId);
    if (!job) return;
    
    this.activeExtractions++;
    
    try {
      // Update job status
      job.status = 'downloading';
      job.updatedAt = new Date();
      this.notifyJobUpdate(job);
      
      // Use native module for youtube-dl integration
      const extractionResult = await YouTubeExtractorModule.extractAudio({
        videoId: job.videoId,
        quality: job.quality,
        format: job.format,
        startTime: job.startTime,
        endTime: job.endTime,
        onProgress: (progress: number) => {
          job.progress = progress;
          job.updatedAt = new Date();
          this.notifyJobUpdate(job);
        },
      });
      
      // Process extracted audio
      job.status = 'processing';
      job.progress = 90;
      this.notifyJobUpdate(job);
      
      const processedAudio = await this.processExtractedAudio(extractionResult);
      
      // Complete job
      job.status = 'completed';
      job.progress = 100;
      job.outputPath = processedAudio.filePath;
      job.fileSize = processedAudio.fileSize;
      job.duration = processedAudio.duration;
      job.videoTitle = extractionResult.title;
      job.attribution.videoTitle = extractionResult.title;
      job.attribution.channelName = extractionResult.channelName;
      job.completedAt = new Date();
      job.updatedAt = new Date();
      
      this.notifyJobUpdate(job);
      
      // Create audio clip record
      await this.createAudioClip(job, processedAudio);
      
    } catch (error) {
      job.status = 'failed';
      job.error = error.message;
      job.updatedAt = new Date();
      this.notifyJobUpdate(job);
    } finally {
      this.activeExtractions--;
      
      // Process next job in queue
      const nextJob = Array.from(this.extractionQueue.values())
        .find(j => j.status === 'pending');
      
      if (nextJob) {
        this.processExtractionJob(nextJob.id);
      }
    }
  }
  
  async cancelExtraction(jobId: string): Promise<void> {
    const job = this.extractionQueue.get(jobId);
    if (!job || job.status === 'completed' || job.status === 'failed') {
      return;
    }
    
    // Cancel native extraction if in progress
    if (job.status === 'downloading' || job.status === 'extracting') {
      await YouTubeExtractorModule.cancelExtraction(jobId);
    }
    
    job.status = 'cancelled';
    job.updatedAt = new Date();
    this.notifyJobUpdate(job);
    
    // Clean up any partial files
    if (job.outputPath) {
      await RNFS.unlink(job.outputPath).catch(() => {});
    }
  }
  
  private async processExtractedAudio(extractionResult: any): Promise<ProcessedAudio> {
    // Normalize audio levels
    const normalizedPath = await AudioProcessorModule.normalizeAudio(extractionResult.filePath);
    
    // Generate waveform data for visualization
    const waveformData = await AudioProcessorModule.generateWaveform(normalizedPath);
    
    // Get file stats
    const stats = await RNFS.stat(normalizedPath);
    
    return {
      filePath: normalizedPath,
      fileSize: stats.size,
      duration: extractionResult.duration,
      waveformData,
    };
  }
  
  private async createAudioClip(job: YouTubeImportJob, processedAudio: ProcessedAudio): Promise<void> {
    const audioClip: YouTubeAudioClip = {
      id: generateId(),
      importJobId: job.id,
      originalVideoId: job.videoId,
      title: job.videoTitle,
      filePath: processedAudio.filePath,
      format: job.format,
      quality: parseInt(job.quality),
      duration: processedAudio.duration,
      fileSize: processedAudio.fileSize,
      waveformData: processedAudio.waveformData,
      metadata: {
        title: job.videoTitle,
        artist: job.attribution.channelName,
      },
      attribution: job.attribution,
      tags: [],
      isOfflineAvailable: true,
      lastAccessedAt: new Date(),
      createdAt: new Date(),
    };
    
    // Save to database
    await YouTubeOfflineService.saveAudioClip(audioClip);
  }
}
```

```typescript
// Copyright compliance and attribution management
class CopyrightComplianceService {
  private copyrightDatabase: Map<string, CopyrightInfo> = new Map();
  
  async checkCopyrightCompliance(videoId: string): Promise<CopyrightCheckResult> {
    // Check against known copyrighted content database
    const copyrightInfo = await this.getCopyrightInfo(videoId);
    
    if (copyrightInfo.isBlocked) {
      return {
        allowed: false,
        reason: 'Content is blocked due to copyright restrictions',
        severity: 'error',
      };
    }
    
    if (copyrightInfo.requiresAttribution) {
      return {
        allowed: true,
        reason: 'Content requires proper attribution',
        severity: 'warning',
        attributionRequired: true,
        attributionText: copyrightInfo.attributionText,
      };
    }
    
    return {
      allowed: true,
      reason: 'Content appears to be safe for personal use',
      severity: 'info',
    };
  }
  
  generateAttributionText(video: YouTubeVideo): string {
    return `"${video.title}" by ${video.channelTitle} (${video.publishedAt.getFullYear()}) - ` +
           `Original: https://youtube.com/watch?v=${video.id} - ` +
           `Used under fair use for personal/educational purposes.`;
  }
  
  async reportCopyrightViolation(videoId: string, reason: string): Promise<void> {
    // Report to internal system and potentially to YouTube
    const report = {
      videoId,
      reason,
      reportedAt: new Date(),
      reportedBy: 'user',
    };
    
    await this.saveCopyrightReport(report);
    
    // Add to blocked content list
    this.copyrightDatabase.set(videoId, {
      isBlocked: true,
      reason: 'User reported copyright violation',
      blockedAt: new Date(),
    });
  }
  
  showCopyrightWarning(video: YouTubeVideo): CopyrightWarning {
    return {
      title: 'Copyright Notice',
      message: `You are about to import audio from "${video.title}" by ${video.channelTitle}. ` +
               'Please ensure you have the right to use this content. ' +
               'This app is for personal and educational use only.',
      guidelines: [
        'Only import content you have permission to use',
        'Respect copyright and fair use guidelines',
        'Provide proper attribution when sharing',
        'Do not use for commercial purposes without permission',
      ],
      acknowledgmentRequired: true,
    };
  }
}
```
