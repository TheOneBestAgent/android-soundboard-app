# Story 3.1: Library Search Integration

## Status
Draft

## Story
**As a** content creator looking for audio clips,
**I want** to search and browse audio libraries (Voicy/MyInstants) with REST API integration,
**so that** I can quickly find and add popular sound effects and memes to my soundboard without leaving the app.

## Acceptance Criteria
1. Search integration with Voicy and MyInstants REST endpoints
2. Search results display within 2 seconds of query submission
3. Audio preview playback before adding to soundboard
4. LRU cache for search results and downloaded audio files
5. Offline support for previously cached search results
6. Download progress indicators with pause/resume functionality
7. Automatic metadata extraction (title, duration, tags, source)
8. Search filters by category, duration, popularity, and source
9. Batch download capability for multiple selected clips
10. Integration with soundboard button assignment workflow

```typescript
// Voicy API integration
class VoicyAPIService implements LibraryAPIService {
  private readonly baseUrl = 'https://api.voicy.network';
  private readonly rateLimiter = new RateLimiter(100, 60000); // 100 requests per minute
  
  async search(query: SearchQuery): Promise<SearchResult[]> {
    await this.rateLimiter.waitForToken();
    
    try {
      const response = await fetch(`${this.baseUrl}/search`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'User-Agent': 'SoundboardApp/1.0',
        },
        body: JSON.stringify({
          query: query.query,
          category: query.category,
          limit: query.limit,
          offset: query.page * query.limit,
        }),
      });
      
      if (!response.ok) {
        throw new Error(`Voicy API error: ${response.status}`);
      }
      
      const data = await response.json();
      return this.normalizeSearchResults(data.results);
    } catch (error) {
      throw new Error(`Voicy search failed: ${error.message}`);
    }
  }
  
  async getDownloadUrl(clipId: string): Promise<string> {
    await this.rateLimiter.waitForToken();
    
    const response = await fetch(`${this.baseUrl}/clips/${clipId}/download`);
    const data = await response.json();
    
    return data.downloadUrl;
  }
  
  private normalizeSearchResults(results: any[]): SearchResult[] {
    return results.map(result => ({
      id: result.id,
      title: result.title,
      description: result.description,
      duration: result.duration,
      previewUrl: result.preview_url,
      downloadUrl: result.download_url,
      source: 'voicy',
      category: result.category,
      tags: result.tags || [],
      popularity: result.play_count || 0,
      uploadDate: new Date(result.created_at),
      fileSize: result.file_size,
      format: result.format,
      metadata: {
        bitrate: result.bitrate,
        sampleRate: result.sample_rate,
        channels: result.channels,
      },
    }));
  }
}

// MyInstants API integration
class MyInstantsAPIService implements LibraryAPIService {
  private readonly baseUrl = 'https://www.myinstants.com/api';
  private readonly rateLimiter = new RateLimiter(50, 60000); // 50 requests per minute
  
  async search(query: SearchQuery): Promise<SearchResult[]> {
    await this.rateLimiter.waitForToken();
    
    try {
      const params = new URLSearchParams({
        q: query.query,
        page: query.page.toString(),
        limit: query.limit.toString(),
      });
      
      const response = await fetch(`${this.baseUrl}/search?${params}`);
      const data = await response.json();
      
      return this.normalizeSearchResults(data.instants);
    } catch (error) {
      throw new Error(`MyInstants search failed: ${error.message}`);
    }
  }
  
  private normalizeSearchResults(results: any[]): SearchResult[] {
    return results.map(result => ({
      id: result.id.toString(),
      title: result.name,
      description: result.description,
      duration: result.duration || 0,
      previewUrl: result.url,
      downloadUrl: result.url,
      source: 'myinstants',
      category: 'general',
      tags: result.tags || [],
      popularity: result.plays || 0,
      uploadDate: new Date(result.created_at),
      fileSize: 0, // Not provided by MyInstants
      format: 'mp3',
      metadata: {},
    }));
  }
}
```

### Caching Strategy
```typescript
// LRU cache implementation for library data
class LibraryCacheManager {
  private searchCache: LRUCache<string, SearchResult[]>;
  private clipCache: LRUCache<string, LibraryClip>;
  private previewCache: LRUCache<string, string>; // URL -> local path
  
  constructor() {
    this.searchCache = new LRUCache({
      max: 1000, // 1000 search queries
      ttl: 1000 * 60 * 30, // 30 minutes
    });
    
    this.clipCache = new LRUCache({
      max: 500, // 500 downloaded clips
      ttl: 1000 * 60 * 60 * 24 * 7, // 1 week
      dispose: this.cleanupClipFile,
    });
    
    this.previewCache = new LRUCache({
      max: 100, // 100 preview files
      ttl: 1000 * 60 * 60, // 1 hour
      dispose: this.cleanupPreviewFile,
    });
  }
  
  async cacheSearchResults(query: string, results: SearchResult[]): Promise<void> {
    const cacheKey = this.generateSearchCacheKey(query);
    this.searchCache.set(cacheKey, results);
    
    // Persist to AsyncStorage for offline access
    await AsyncStorage.setItem(
      `search_cache_${cacheKey}`,
      JSON.stringify(results)
    );
  }
  
  async getCachedSearchResults(query: string): Promise<SearchResult[] | null> {
    const cacheKey = this.generateSearchCacheKey(query);
    
    // Try memory cache first
    let results = this.searchCache.get(cacheKey);
    
    if (!results) {
      // Try persistent cache
      const cached = await AsyncStorage.getItem(`search_cache_${cacheKey}`);
      if (cached) {
        results = JSON.parse(cached);
        this.searchCache.set(cacheKey, results);
      }
    }
    
    return results || null;
  }
  
  async cacheClip(clip: LibraryClip): Promise<void> {
    this.clipCache.set(clip.id, clip);
    
    // Update access time
    clip.lastAccessed = new Date();
    
    // Persist to database
    await this.persistClipToDatabase(clip);
  }
  
  private generateSearchCacheKey(query: string): string {
    return btoa(query).replace(/[^a-zA-Z0-9]/g, '');
  }
  
  private async cleanupClipFile(clip: LibraryClip): Promise<void> {
    try {
      await RNFS.unlink(clip.localPath);
    } catch (error) {
      console.warn('Failed to cleanup clip file:', error);
    }
  }
}
```

### Download Management
```typescript
// Download queue with progress tracking
class DownloadManager {
  private downloadQueue: DownloadJob[] = [];
  private activeDownloads: Map<string, DownloadJob> = new Map();
  private maxConcurrentDownloads = 3;
  
  async downloadClip(searchResult: SearchResult): Promise<string> {
    const job: DownloadJob = {
      id: generateId(),
      searchResult,
      status: 'queued',
      progress: 0,
      startTime: Date.now(),
      localPath: this.generateLocalPath(searchResult),
    };
    
    this.downloadQueue.push(job);
    this.processQueue();
    
    return new Promise((resolve, reject) => {
      job.onComplete = resolve;
      job.onError = reject;
    });
  }
  
  async downloadBatch(searchResults: SearchResult[]): Promise<string[]> {
    const downloadPromises = searchResults.map(result => 
      this.downloadClip(result)
    );
    
    return Promise.all(downloadPromises);
  }
  
  private async processQueue(): Promise<void> {
    while (
      this.downloadQueue.length > 0 && 
      this.activeDownloads.size < this.maxConcurrentDownloads
    ) {
      const job = this.downloadQueue.shift()!;
      this.activeDownloads.set(job.id, job);
      
      this.executeDownload(job).finally(() => {
        this.activeDownloads.delete(job.id);
        this.processQueue();
      });
    }
  }
  
  private async executeDownload(job: DownloadJob): Promise<void> {
    try {
      job.status = 'downloading';
      
      const downloadResult = await RNFS.downloadFile({
        fromUrl: job.searchResult.downloadUrl,
        toFile: job.localPath,
        progress: (res) => {
          job.progress = (res.bytesWritten / res.contentLength) * 100;
          this.notifyProgress(job);
        },
      }).promise;
      
      if (downloadResult.statusCode === 200) {
        job.status = 'completed';
        job.completedAt = Date.now();
        
        // Cache the downloaded clip
        const clip: LibraryClip = {
          id: generateId(),
          searchResultId: job.searchResult.id,
          localPath: job.localPath,
          metadata: job.searchResult,
          downloadDate: new Date(),
          lastAccessed: new Date(),
          isFavorite: false,
          customTags: [],
          assignedButtons: [],
        };
        
        await LibraryCacheManager.getInstance().cacheClip(clip);
        
        job.onComplete?.(job.localPath);
      } else {
        throw new Error(`Download failed with status ${downloadResult.statusCode}`);
      }
    } catch (error) {
      job.status = 'failed';
      job.error = error.message;
      job.onError?.(error);
    }
  }
  
  pauseDownload(jobId: string): void {
    const job = this.activeDownloads.get(jobId);
    if (job) {
      job.status = 'paused';
      // Implementation depends on download library capabilities
    }
  }
  
  resumeDownload(jobId: string): void {
    const job = this.activeDownloads.get(jobId);
    if (job && job.status === 'paused') {
      job.status = 'downloading';
      // Resume download implementation
    }
  }
}
```
